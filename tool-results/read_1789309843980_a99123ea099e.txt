    53→export function streakAt(streak: number): KillstreakDef | null {
    54→  let best: KillstreakDef | null = null
    55→  for (const k of KILLSTREAKS) if (streak >= k.streak && (!best || k.streak > best.streak)) best = k
    56→  return best
    57→}

    98→// Build a house: 4 walls with a door gap on one side, windows, optional roof
    99→// Returns array of obstacles (wall segments)
   100→function buildHouse(cx: number, cz: number, w: number, d: number, h: number, color: number, roofColor: number, doorSide: 'N'|'S'|'E'|'W' = 'S'): MapObstacle[] {
   101→  const walls: MapObstacle[] = []
   102→  const wallT = 0.3
   103→  const doorW = 1.5
   104→  const doorH = 2.2  // door height
   105→  const winH = 1.0
   106→  // North wall (z = cz - d/2)
   107→  if (doorSide !== 'N') {
   108→    walls.push(ob(cx, cz - d/2, w, h, wallT, false, color, 'wall'))
   109→  } else {
   110→    walls.push(ob(cx - w/4 - doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall'))
   111→    walls.push(ob(cx + w/4 + doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall'))
   112→    // window above door (positioned at doorH, no collision)
   113→    walls.push(ob(cx, cz - d/2, doorW, h - doorH, wallT, false, color, 'wall', 0, true, doorH))
   114→  }
   115→  // South wall (z = cz + d/2)
   116→  if (doorSide !== 'S') {
   117→    walls.push(ob(cx, cz + d/2, w, h, wallT, false, color, 'wall'))
   118→  } else {
   119→    walls.push(ob(cx - w/4 - doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall'))
   120→    walls.push(ob(cx + w/4 + doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall'))
   121→    walls.push(ob(cx, cz + d/2, doorW, h - doorH, wallT, false, color, 'wall', 0, true, doorH))
   122→  }
   123→  // East wall (x = cx + w/2)
   124→  if (doorSide !== 'E') {
   125→    walls.push(ob(cx + w/2, cz, wallT, h, d, false, color, 'wall'))
   126→  } else {
   127→    walls.push(ob(cx + w/2, cz - d/4 - doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall'))
   128→    walls.push(ob(cx + w/2, cz + d/4 + doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall'))
   129→  }
   130→  // West wall (x = cx - w/2)
   131→  if (doorSide !== 'W') {
   132→    walls.push(ob(cx - w/2, cz, wallT, h, d, false, color, 'wall'))
   133→  } else {
   134→    walls.push(ob(cx - w/2, cz - d/4 - doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall'))
   135→    walls.push(ob(cx - w/2, cz + d/4 + doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall'))
   136→  }
   137→  // Roof at top of walls (player can stand on it)
   138→  walls.push(ob(cx, cz, w + 0.6, 0.3, d + 0.6, true, roofColor, 'roof', 0, false, h))
   139→  return walls
   140→}

   142→// Build stairs: a series of steps going up, each at increasing height
   143→function buildStairs(cx: number, cz: number, steps: number, dir: 'N'|'S'|'E'|'W', color: number): MapObstacle[] {
   144→  const result: MapObstacle[] = []
   145→  const stepH = 0.6, stepD = 0.7, stepW = 2.5
   146→  for (let i = 0; i < steps; i++) {
   147→    let x = cx, z = cz
   148→    if (dir === 'N') z = cz - i * stepD
   149→    if (dir === 'S') z = cz + i * stepD
   150→    if (dir === 'E') x = cx + i * stepD
   151→    if (dir === 'W') x = cx - i * stepD
   152→    // each step is positioned at y = i * stepH (stacking upward)
   153→    result.push(ob(x, z, stepW, stepH, stepD, true, color, 'stair', 0, false, i * stepH))
   154→  }
   155→  return result
   156→}

   158→// Build a tree (trunk + foliage on top)
   159→function buildTree(cx: number, cz: number, scale: number = 1): MapObstacle[] {
   160→  const trunkH = 3 * scale
   161→  const foliageH = 2.5 * scale
   162→  return [
   163→    // trunk at ground level
   164→    ob(cx, cz, 0.4 * scale, trunkH, 0.4 * scale, false, 0x7a5230, 'cyl'),
   165→    // foliage on top of trunk
   166→    ob(cx, cz, 2.5 * scale, foliageH, 2.5 * scale, true, 0x27ae60, 'box', 0, false, trunkH),
   167→  ]
   168→}
   169→
   170→// Build a car (body + cabin on top)
   171→function buildCar(cx: number, cz: number, color: number, rotation: number = 0): MapObstacle[] {
   172→  return [
   173→    ob(cx, cz, 2, 0.8, 4, true, color, 'box', rotation),           // body at ground
   174→    ob(cx, cz, 1.8, 0.6, 2, false, 0x2c3e50, 'box', rotation, false, 0.8), // cabin on top
   175→  ]
   176→}

   178→// Build a multi-floor tower: enterable, with internal stairs
   179→// floors = number of stories, each floor has walls with a door + stairs inside
   180→function buildTower(cx: number, cz: number, w: number, d: number, floors: number, color: number, roofColor: number): MapObstacle[] {
   181→  const result: MapObstacle[] = []
   182→  const floorH = 3.0  // height per floor
   183→  const wallT = 0.3
   184→  const doorW = 1.5
   185→  const totalH = floors * floorH
   186→  // Outer walls (full height, with door gap on south side at ground floor only)
   187→  // North wall (full)
   188→  result.push(ob(cx, cz - d/2, w, totalH, wallT, false, color, 'wall'))
   189→  // East wall (full)
   190→  result.push(ob(cx + w/2, cz, wallT, totalH, d, false, color, 'wall'))
   191→  // West wall (full)
   192→  result.push(ob(cx - w/2, cz, wallT, totalH, d, false, color, 'wall'))
   193→  // South wall with door gaps at each floor
   194→  for (let f = 0; f < floors; f++) {
   195→    const fy = f * floorH
   196→    // wall segments on either side of door
   197→    result.push(ob(cx - w/4 - doorW/4, cz + d/2, w/2 - doorW/2, floorH - 0.1, wallT, false, color, 'wall', 0, false, fy))
   198→    result.push(ob(cx + w/4 + doorW/4, cz + d/2, w/2 - doorW/2, floorH - 0.1, wallT, false, color, 'wall', 0, false, fy))
   199→    // window above door
   200→    result.push(ob(cx, cz + d/2, doorW, 0.8, wallT, false, color, 'wall', 0, true, fy + 2.0))
   201→    // floor platform (roof type, walkable) for upper floors
   202→    if (f > 0) {
   203→      result.push(ob(cx, cz, w - wallT * 2, 0.2, d - wallT * 2, true, color, 'roof', 0, false, fy))
   204→      // stairs inside (going up to next floor)
   205→      if (f < floors) {
   206→        result.push(...buildStairs(cx - w/4, cz, 4, 'N', 0x8a7a5a).map(s => ({ ...s, y: (s.y || 0) + fy })))
   207→      }
   208→    }
   209→  }
   210→  // Roof at top
   211→  result.push(ob(cx, cz, w + 0.6, 0.3, d + 0.6, true, roofColor, 'roof', 0, false, totalH))
   212→  return result
   213→}
