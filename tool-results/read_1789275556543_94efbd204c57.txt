   205→function pveLevelConfig(level: number) {
   206→  const baseHp = 50
   207→  return {
   208→    mobCount: Math.min(14, 3 + Math.floor(level * 1.2)),
   209→    mobHp: baseHp + (level - 1) * 15,
   210→    mobSpeed: Math.min(6.0, 2.8 + (level - 1) * 0.25),
   211→    mobDamage: 6 + (level - 1) * 1.5,
   212→    mapId: MAPS[(level - 1) % MAPS.length].id,
   213→  }
   214→}

   216→const KILLSTREAKS = [
   217→  { id: 'drone', name: 'Dron', streak: 3, desc: 'Dron aliado que ataca enemigos', icon: 'drone' },
   218→  { id: 'bomb',  name: 'Bomba', streak: 5, desc: 'Bombardeo de zona', icon: 'bomb' },

   221→function streakAt(streak: number) {
   222→  let best: typeof KILLSTREAKS[0] | null = null
   223→  for (const k of KILLSTREAKS) if (streak >= k.streak && (!best || k.streak > best.streak)) best = k
   224→  return best
   225→}

   227→function dist2D(a: [number, number, number], b: [number, number, number]) {
   228→  const dx = a[0] - b[0], dz = a[2] - b[2]
   229→  return Math.sqrt(dx * dx + dz * dz)
   230→}
   231→function clampToArena(p: [number, number, number]) {
   232→  const lim = HALF - WALL - 0.5
   233→  p[0] = Math.max(-lim, Math.min(lim, p[0]))
   234→  p[2] = Math.max(-lim, Math.min(lim, p[2]))
   235→}

   307→function spawnBlocked(map: GameMap, x: number, z: number): boolean {
   308→  // check if a position is inside any obstacle (with margin)
   309→  const margin = 1.5
   310→  for (const o of map.obstacles) {
   311→    if (Math.abs(x - o.x) < o.w/2 + margin && Math.abs(z - o.z) < o.d/2 + margin) return true
   312→  }
   313→  return false
   314→}

   316→function pickSpawn(room: Room, preferTeam: Team = 'none'): [number, number, number] {
   317→  const map = getMap(room.mapId)
   318→  const spawns = map.spawns
   319→  // try each spawn with random offset, validate not inside obstacle
   320→  const trySpawn = (s: [number, number]): [number, number, number] => {
   321→    for (let attempt = 0; attempt < 8; attempt++) {
   322→      const x = s[0] + (Math.random() - 0.5) * 4
   323→      const z = s[1] + (Math.random() - 0.5) * 4
   324→      if (!spawnBlocked(map, x, z)) return [x, 1.7, z]
   325→    }
   326→    // fallback: use exact spawn point
   327→    return [s[0], 1.7, s[1]]
   328→  }
   329→  // PvP: bias by team (blue near spawn[0], red near spawn[1])
   330→  if (preferTeam !== 'none' && spawns.length >= 2) {
   331→    const idx = preferTeam === 'blue' ? 0 : 1
   332→    return trySpawn(spawns[idx])
   333→  }
   334→  // pick spawn farthest from any alive player
   335→  let best = spawns[0], bestD = -1
   336→  for (const s of spawns) {
   337→    let minD = Infinity
   338→    for (const p of room.players.values()) {
   339→      if (p.state !== 'alive') continue
   340→      minD = Math.min(minD, dist2D([s[0], 0, s[1]], p.pos))
   341→    }
   342→    if (minD > bestD) { bestD = minD; best = s }
   343→  }
   344→  return trySpawn(best)
   345→}

   347→function makePlayer(socketId: string, name: string, skin: string, room: Room): Player {
   348→  // team assignment for PvP

   512→function checkPvERoundEnd(r: Room) {
   513→  if (!r.roundActive) return
   514→  const aliveMobs = r.mobs.filter(m => m.state === 'alive').length
   515→  if (aliveMobs === 0) {
   516→    r.roundActive = false
   517→    r.roundEndsAt = Date.now() + ROUND_END_MS
   518→    r.nextLevel = r.level + 1
   519→    io.to(r.id).emit('room:mapChange', { mapId: r.mapId, level: r.level, mode: r.mode, banner: `¡Nivel ${r.level} superado!` })
   520→  }

   523→function checkPvPRoundEnd(r: Room) {
   524→  if (!r.roundActive || r.players.size < 2) return
   525→  const teams = { blue: 0, red: 0 }
   526→  for (const p of r.players.values()) {
   527→    if (p.state !== 'alive') continue
   528→    if (p.team === 'blue' || p.team === 'red') teams[p.team]++
   529→  }
   530→  if (teams.blue === 0 || teams.red === 0) {
   531→    r.roundActive = false
   532→    r.roundEndsAt = Date.now() + ROUND_END_MS
   533→    const winner = teams.blue > 0 ? 'Azul' : 'Rojo'
   534→    io.to(r.id).emit('room:mapChange', { mapId: r.mapId, level: r.level, mode: r.mode, banner: `¡Equipo ${winner} gana la ronda!` })
   535→  }

   735→    console.log(`[disconnect] ${socket.id}`)
   736→    leaveRoom(socket)
   737→    socketToLobby.delete(socket.id)
   738→  })
   739→  socket.on('error', (err) => console.error(`[socket error] ${socket.id}`, err))

   742→function applyDamageToPlayer(room: Room, target: Player, dmg: number, killerId: string, weapon: string, headshot: boolean, attackerPos?: [number, number, number]) {
   743→  const now = Date.now()
   744→  target.lastDamagedAt = now
   745→  // shield absorbs first
   746→  let remaining = dmg
   747→  if (target.shield > 0) {
   748→    const absorbed = Math.min(target.shield, remaining * 0.6)
   749→    target.shield -= absorbed
   750→    remaining -= absorbed
   751→  }
   752→  target.health -= remaining
   753→  io.to(room.id).emit('player:damaged', {
   754→    id: target.id, health: Math.max(0, target.health),
   755→    shield: Math.max(0, target.shield), by: killerId, headshot,
   756→    attackerPos: attackerPos ?? null,
   757→  })
   758→  if (target.health <= 0) killPlayer(room, target, killerId, weapon, headshot)
   759→}

   761→function joinRoom(socket: any, room: Room, name: string, skin: string) {
   762→  leaveRoom(socket)
   763→  const player = makePlayer(socket.id, name, skin, room)
   764→  room.players.set(socket.id, player)
   765→  socketToRoom.set(socket.id, room.id)
   766→  socket.join(room.id)
   767→  socket.emit('room:joined', {
   768→    room: { id: room.id, name: room.name, mode: room.mode, mapId: room.mapId, level: room.level },
   769→    me: playerPublic(player),
   770→    players: roomRoster(room),
   771→    mobs: room.mobs.map(m => ({ id: m.id, pos: m.pos, state: m.state, health: m.health })),
   772→    items: room.items.map(i => ({ id: i.id, type: i.type, pos: i.pos })),
   773→  })
   774→  socket.to(room.id).emit('room:playerJoined', playerPublic(player))
   775→  room.emptySince = null
   776→  broadcastLobbyState()
   777→  console.log(`[join] ${player.name} -> ${room.name} (${room.mode}, ${room.players.size}p)`)
   778→}

   780→function leaveRoom(socket: any) {
   781→  const rid = socketToRoom.get(socket.id)
   782→  if (!rid) return
   783→  const room = rooms.get(rid)
   784→  socketToRoom.delete(socket.id)
   785→  socket.leave(rid)
   786→  if (room) {
   787→    const p = room.players.get(socket.id)
   788→    room.players.delete(socket.id)
   789→    if (p) {
   790→      p.connected = false
   791→      socket.to(room.id).emit('room:playerLeft', { id: socket.id })
   792→    }
   793→    if (room.players.size === 0) {
   794→      room.emptySince = Date.now()
   795→    }
   796→  }
   797→  broadcastLobbyState()
   798→}

   800→function killPlayer(room: Room, victim: Player, killerId: string, weapon: string, headshot: boolean) {
   801→  victim.state = 'dead'
   802→  victim.health = 0
   803→  victim.deaths += 1
   804→  victim.streak = 0
   805→  victim.respawnAt = Date.now() + RESPAWN_MS
   806→  const killer = room.players.get(killerId)
   807→  if (killer && killer.id !== victim.id) {
   808→    killer.kills += 1
   809→    killer.streak += 1
   810→    killer.bestStreak = Math.max(killer.bestStreak, killer.streak)
   811→    killer.score += headshot ? 150 : 100
   812→    // streak rewards
   813→    const reward = streakAt(killer.streak)
   814→    if (reward && killer.streak === reward.streak) {
   815→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
   816→      io.to(killer.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
   817→      applyStreakReward(room, killer, reward.id)
   818→    } else {
   819→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward: null })
   820→    }
   821→    // PvP: drop a random item on death
   822→    if (room.mode === 'pvp' && Math.random() < 0.6) dropItemOnMobKill(room, victim.pos)
   823→  }
   824→  io.to(room.id).emit('player:killed', { victimId: victim.id, killerId, weapon, headshot })
   825→  killFeed(room, victim.name, killer?.name ?? null, weapon, headshot, 'player')
   826→  setTimeout(() => {
   827→    if (!victim.connected) return
   828→    if (!room.players.has(victim.id)) return
   829→    // Always respawn — round restarts will override with fresh state if needed
   830→    victim.state = 'alive'
   831→    victim.health = PLAYER_MAX_HP
   832→    victim.shield = 0
   833→    victim.pos = pickSpawn(room, victim.team)
   834→    victim.weapon = 'pistol'
   835→    victim.ammo = WEAPONS.pistol.magazine
   836→    victim.reloading = false
   837→    io.to(room.id).emit('player:respawned', { id: victim.id, pos: victim.pos })
   838→  }, RESPAWN_MS)
   839→  // round-end checks
   840→  if (room.mode === 'pvp') checkPvPRoundEnd(room)
   841→}

   843→function applyStreakReward(room: Room, p: Player, rewardId: string) {
   844→  if (rewardId === 'heal' || rewardId === 'aura') {
   845→    p.health = Math.min(PLAYER_MAX_HP, p.health + 25)
   846→    p.shield = Math.min(PLAYER_MAX_SHIELD, p.shield + 25)
   847→  }
   848→  if (rewardId === 'bomb') {
   849→    // damage all mobs near player + nearby enemies
   850→    for (const m of room.mobs) {
   851→      if (m.state !== 'alive') continue
   852→      if (dist2D(m.pos, p.pos) < 12) {
   853→        m.health -= 80
   854→        m.hitFlash = Date.now()
   855→        io.to(room.id).emit('mob:damaged', { id: m.id, health: Math.max(0, m.health), by: p.id })
   856→        if (m.health <= 0) killMob(room, m, p.id, 'bomb', false)
   857→      }
   858→    }
   859→    for (const other of room.players.values()) {
   860→      if (other.id === p.id || other.state !== 'alive') continue
   861→      if (room.mode === 'pvp' && other.team === p.team && p.team !== 'none') continue
   862→      if (dist2D(other.pos, p.pos) < 10) {
   863→        applyDamageToPlayer(room, other, 60, p.id, 'bomb', false)
   864→      }
   865→    }
   866→  }
   867→  if (rewardId === 'drone') {
   868→    room.droneOwner = p.id
   869→    room.droneEnd = Date.now() + 20000
   870→    io.to(room.id).emit('drone:state', { ownerId: p.id, pos: [p.pos[0], 3.5, p.pos[2]], expires: room.droneEnd, targetYaw: p.yaw })
   871→  }
   872→  if (rewardId === 'aura') {
   873→    p.ammo = WEAPONS[p.weapon].magazine
   874→    p.reloading = false
   875→    io.to(p.id).emit('player:reloadDone', { weapon: p.weapon, ammo: p.ammo })
   876→  }
   877→}

   879→function killMob(room: Room, mob: Mob, killerId: string, weapon: string, headshot: boolean) {
   880→  mob.state = 'dead'
   881→  mob.health = 0
   882→  mob.respawnAt = Date.now() + MOB_RESPAWN_MS
   883→  const killer = room.players.get(killerId)
   884→  if (killer) {
   885→    killer.streak += 1
   886→    killer.bestStreak = Math.max(killer.bestStreak, killer.streak)
   887→    killer.score += headshot ? 60 : 40
   888→    const reward = streakAt(killer.streak)
   889→    if (reward && killer.streak === reward.streak) {
   890→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
   891→      io.to(killer.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
   892→      applyStreakReward(room, killer, reward.id)
   893→    } else {
   894→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward: null })
   895→    }
   896→  }
   897→  io.to(room.id).emit('mob:killed', { id: mob.id, killerId, weapon, headshot })
   898→  killFeed(room, 'Doodle Mob', killer?.name ?? null, weapon, headshot, 'mob')
   899→  // item drop
   900→  dropItemOnMobKill(room, mob.pos)
   901→  checkPvERoundEnd(room)
   902→  // schedule respawn
   903→  setTimeout(() => {
   904→    if (!rooms.has(room.id)) return
   905→    if (room.mode !== 'pve') return // mobs only respawn in PvE between levels (handled by round restart)
   906→    mob.state = 'alive'
   907→    mob.health = mob.maxHealth
   908→    const a = Math.random() * Math.PI * 2
   909→    const r = Math.random() * (HALF - 8)
   910→    mob.pos = [Math.cos(a) * r, 1, Math.sin(a) * r]
   911→    mob.wanderTarget = mob.pos.slice() as [number, number, number]
   912→    io.to(room.id).emit('mob:respawned', { id: mob.id, pos: mob.pos })
   913→  }, MOB_RESPAWN_MS)
   914→}

   916→// ----------------------- World tick -----------------------
   917→setInterval(() => {
   918→  const now = Date.now()
   919→  // cleanup empty rooms
   920→  for (const [rid, r] of rooms) {
   921→    if (r.players.size === 0) {
   922→      if (r.emptySince == null) r.emptySince = now
   923→      if (now - (r.emptySince as number) > EMPTY_ROOM_TTL_MS) {
   924→        // keep at least the default PvE room
   925→        if (rooms.size > 1) {
   926→          rooms.delete(rid)
   927→          continue
   928→        }
   929→      }
   930→    }
   931→  }
   932→
   933→  for (const room of rooms.values()) {
   934→    if (room.players.size === 0) continue
   935→
   936→    // round transitions
   937→    if (!room.roundActive && room.roundEndsAt && now >= room.roundEndsAt) {
   938→      room.roundEndsAt = 0
   939→      if (room.mode === 'pve') startPvERound(room, room.nextLevel)
   940→      else startPvPRound(room, true)
   941→    }
   942→
   943→    // mob AI (PvE)
   944→    if (room.mode === 'pve') {
   945→      const cfg = pveLevelConfig(room.level)
   946→      for (const mob of room.mobs) {
   947→        if (mob.state === 'dead') continue
   948→        let nearest: Player | null = null
   949→        let nd = Infinity
   950→        for (const p of room.players.values()) {
   951→          if (p.state !== 'alive') continue
   952→          const d = dist2D(mob.pos, p.pos)
   953→          if (d < nd) { nd = d; nearest = p }
   954→        }
   955→        let tx: number, tz: number
   956→        if (nearest && nd < 16) {
   957→          tx = nearest.pos[0]; tz = nearest.pos[2]
   958→          if (nd < 1.9 && now - mob.lastAttack > MOB_ATTACK_CD) {
   959→            mob.lastAttack = now
   960→            applyDamageToPlayer(room, nearest, cfg.mobDamage, mob.id, 'mob', false, mob.pos)
   961→          }
   962→        } else {
   963→          if (dist2D(mob.pos, mob.wanderTarget) < 1.5) {
   964→            const a = Math.random() * Math.PI * 2
   965→            const r = Math.random() * (HALF - 6)
   966→            mob.wanderTarget = [Math.cos(a) * r, 1, Math.sin(a) * r]
   967→          }
   968→          tx = mob.wanderTarget[0]; tz = mob.wanderTarget[2]
   969→        }
   970→        const dx = tx - mob.pos[0], dz = tz - mob.pos[2]
   971→        const len = Math.hypot(dx, dz) || 1
   972→        const vx = (dx / len) * cfg.mobSpeed, vz = (dz / len) * cfg.mobSpeed
   973→        mob.vel[0] = vx; mob.vel[1] = vz
   974→        const step = TICK_MS / 1000
   975→        mob.pos[0] += vx * step
   976→        mob.pos[2] += vz * step
   977→        clampToArena(mob.pos)
   978→        mob.pos[1] = 1
   979→      }
   980→      io.to(room.id).emit('mob:state', { mobs: room.mobs.map(m => ({ id: m.id, pos: m.pos, state: m.state, flash: m.hitFlash })) })
   981→    }
   982→
   983→    // PvE health regen: if player hasn't taken damage for 5s, regen 1.5 HP/sec
   984→    if (room.mode === 'pve') {
   985→      for (const p of room.players.values()) {
   986→        if (p.state !== 'alive') continue
   987→        if (p.health >= PLAYER_MAX_HP) continue
   988→        if (now - p.lastDamagedAt > 5000) {
   989→          p.health = Math.min(PLAYER_MAX_HP, p.health + 1.5 * (TICK_MS / 1000) * 4)
   990→          // send updated health to the player (throttled)
   991→          if (now % 500 < TICK_MS) {
   992→            io.to(p.id).emit('player:damaged', { id: p.id, health: Math.round(p.health), shield: p.shield, by: 'regen', headshot: false, attackerPos: null, regen: true })
   993→          }
   994→        }
   995→      }
   996→    }
   997→
   998→    // drone streak reward: auto-attack nearest enemy/mob
   999→    if (room.droneOwner && now < room.droneEnd) {
  1000→      const owner = room.players.get(room.droneOwner)
  1001→      if (owner && owner.state === 'alive') {
  1002→        // attack nearest mob every ~700ms
  1003→        if (now % 700 < TICK_MS) {
  1004→          let target: Mob | null = null; let td = Infinity
  1005→          for (const m of room.mobs) {
  1006→            if (m.state !== 'alive') continue
  1007→            const d = dist2D(m.pos, owner.pos)
  1008→            if (d < 18 && d < td) { td = d; target = m }
  1009→          }
  1010→          if (target) {
  1011→            target.health -= 14
  1012→            target.hitFlash = now
  1013→            io.to(room.id).emit('mob:damaged', { id: target.id, health: Math.max(0, target.health), by: owner.id })
  1014→            if (target.health <= 0) killMob(room, target, owner.id, 'drone', false)
  1015→          }
  1016→        }
  1017→      } else if (owner && owner.state !== 'alive') {
  1018→        // drone persists but does nothing while dead
  1019→      } else {
  1020→        room.droneOwner = null
  1021→      }
  1022→    } else if (room.droneOwner && now >= room.droneEnd) {
  1023→      room.droneOwner = null
  1024→      io.to(room.id).emit('drone:state', { ownerId: null, pos: null, expires: 0 })
  1025→    }
  1026→    // emit drone position to all clients (for 3D rendering)
  1027→    if (room.droneOwner && now < room.droneEnd) {
  1028→      const owner = room.players.get(room.droneOwner)
  1029→      if (owner && owner.state === 'alive') {
  1030→        // drone hovers above and behind the owner
  1031→        const dx = -Math.sin(owner.yaw) * 3
  1032→        const dz = -Math.cos(owner.yaw) * 3
  1033→        const dronePos: [number, number, number] = [owner.pos[0] + dx, 3.5, owner.pos[2] + dz]
  1034→        io.to(room.id).emit('drone:state', { ownerId: room.droneOwner, pos: dronePos, expires: room.droneEnd, targetYaw: owner.yaw })
  1035→      }
  1036→    }
  1037→
  1038→    // item expiry
  1039→    let itemsChanged = false
  1040→    room.items = room.items.filter(it => {
  1041→      if (now - it.born > ITEM_LIFETIME_MS) { itemsChanged = true; return false }
  1042→      return true
  1043→    })
  1044→    if (itemsChanged) {
  1045→      io.to(room.id).emit('items:state', { items: room.items.map(i => ({ id: i.id, type: i.type, pos: i.pos })) })
  1046→    }
  1047→
  1048→    // periodic item spawns (PvE: every ~12s, PvP: every ~16s)
  1049→    room.itemSpawnTimer += TICK_MS
  1050→    const interval = room.mode === 'pve' ? ITEM_RESPAWN_INTERVAL_MS : 16000
  1051→    if (room.itemSpawnTimer >= interval && room.items.length < 6) {
  1052→      room.itemSpawnTimer = 0
  1053→      const roll = Math.random()
  1054→      const type: 'ammo'|'heal'|'shield' = roll < 0.5 ? 'ammo' : roll < 0.8 ? 'heal' : 'shield'
  1055→      spawnItem(room, type)
  1056→    }
  1057→
  1058→    // roster every ~1s
  1059→    if (now % 1000 < TICK_MS) emitRoomState(room)
  1060→  }
  1061→}, TICK_MS)
