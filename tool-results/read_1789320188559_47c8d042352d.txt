    56→function buildHouse(cx: number, cz: number, w: number, d: number, h: number, color: number, roofColor: number, doorSide: 'N'|'S'|'E'|'W' = 'S'): MapObstacle[] {
    57→  const walls: MapObstacle[] = []
    58→  const wallT = 0.3
    59→  const doorW = 1.5
    60→  const doorH = 2.2  // door height
    61→  if (doorSide !== 'N') { walls.push(ob(cx, cz - d/2, w, h, wallT, false, color, 'wall')) }
    62→  else { walls.push(ob(cx - w/4 - doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx + w/4 + doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx, cz - d/2, doorW, h - doorH, wallT, false, color, 'wall', 0, true, doorH)) }
    63→  if (doorSide !== 'S') { walls.push(ob(cx, cz + d/2, w, h, wallT, false, color, 'wall')) }
    64→  else { walls.push(ob(cx - w/4 - doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx + w/4 + doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx, cz + d/2, doorW, h - doorH, wallT, false, color, 'wall', 0, true, doorH)) }
    65→  if (doorSide !== 'E') { walls.push(ob(cx + w/2, cz, wallT, h, d, false, color, 'wall')) }
    66→  else { walls.push(ob(cx + w/2, cz - d/4 - doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')); walls.push(ob(cx + w/2, cz + d/4 + doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')) }
    67→  if (doorSide !== 'W') { walls.push(ob(cx - w/2, cz, wallT, h, d, false, color, 'wall')) }
    68→  else { walls.push(ob(cx - w/2, cz - d/4 - doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')); walls.push(ob(cx - w/2, cz + d/4 + doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')) }
    69→  walls.push(ob(cx, cz, w + 0.6, 0.3, d + 0.6, true, roofColor, 'roof', 0, false, h))
    70→  return walls
    71→}

    73→function buildStairs(cx: number, cz: number, steps: number, dir: 'N'|'S'|'E'|'W', color: number): MapObstacle[] {
    74→  const result: MapObstacle[] = []
    75→  const stepH = 0.6, stepD = 0.7, stepW = 2.5
    76→  for (let i = 0; i < steps; i++) { let x = cx, z = cz; if (dir === 'N') z = cz - i * stepD; if (dir === 'S') z = cz + i * stepD; if (dir === 'E') x = cx + i * stepD; if (dir === 'W') x = cx - i * stepD; result.push(ob(x, z, stepW, stepH, stepD, true, color, 'stair', 0, false, i * stepH)) }
    77→  return result
    78→}

    80→function buildTree(cx: number, cz: number, scale: number = 1): MapObstacle[] {
    81→  const trunkH = 3 * scale
    82→  const foliageH = 2.5 * scale
    83→  return [
    84→    ob(cx, cz, 0.4 * scale, trunkH, 0.4 * scale, false, 0x7a5230, 'cyl'),
    85→    ob(cx, cz, 2.5 * scale, foliageH, 2.5 * scale, true, 0x27ae60, 'box', 0, false, trunkH),
    86→  ]
    87→}

    96→function buildTower(cx: number, cz: number, w: number, d: number, floors: number, color: number, roofColor: number): MapObstacle[] {
    97→  const result: MapObstacle[] = []
    98→  const floorH = 3.0
    99→  const wallT = 0.3
   100→  const doorW = 1.5
   101→  const totalH = floors * floorH
   102→  // Outer walls (full height) — North, East, West
   103→  result.push(ob(cx, cz - d/2, w, totalH, wallT, false, color, 'wall'))
   104→  result.push(ob(cx + w/2, cz, wallT, totalH, d, false, color, 'wall'))
   105→  result.push(ob(cx - w/2, cz, wallT, totalH, d, false, color, 'wall'))
   106→  // South wall with door gap at ground floor only
   107→  result.push(ob(cx - w/4 - doorW/4, cz + d/2, w/2 - doorW/2, floorH - 0.1, wallT, false, color, 'wall', 0, false, 0))
   108→  result.push(ob(cx + w/4 + doorW/4, cz + d/2, w/2 - doorW/2, floorH - 0.1, wallT, false, color, 'wall', 0, false, 0))
   109→  // Upper floor walls (full)
   110→  for (let f = 1; f < floors; f++) {
   111→    const fy = f * floorH
   112→    result.push(ob(cx, cz + d/2, w, floorH - 0.1, wallT, false, color, 'wall', 0, false, fy))
   113→  }
   114→  // Floor platforms for upper floors (walkable, thin slabs)
   115→  for (let f = 1; f < floors; f++) {
   116→    const fy = f * floorH
   117→    result.push(ob(cx + w/4, cz, w/2 - wallT, 0.2, d - wallT * 2, true, color, 'roof', 0, false, fy))
   118→  }
   119→  // Stairs: start from floor 1 (ground) going up to each floor
   120→  for (let f = 0; f < floors - 1; f++) {
   121→    const fy = f * floorH
   122→    result.push(...buildStairs(cx, cz - d/4, 5, 'N', 0x8a7a5a).map(s => ({ ...s, y: (s.y || 0) + fy })))
   123→  }
   124→  // Roof at top
   125→  result.push(ob(cx, cz, w + 0.6, 0.3, d + 0.6, true, roofColor, 'roof', 0, false, totalH))
   126→  return result
   127→}

   497→function spawnBlocked(map: GameMap, x: number, z: number): boolean {
   498→  // check if a position is inside any obstacle (with margin)
   499→  // skip water and noCollide obstacles (they don't block spawning)
   500→  // skip elevated obstacles (y > 1.0): they float above ground and don't block ground spawns
   501→  const margin = 1.5
   502→  for (const o of map.obstacles) {
   503→    if (o.kind === 'water' || o.noCollide) continue
   504→    if ((o.y ?? 0) > 1.0) continue
   505→    const hw = o.w / 2, hd = o.d / 2
   506→    if (o.rotation && o.rotation !== 0) {
   507→      // rotated: use slightly larger margin (bounding-circle approximation)
   508→      const reach = Math.max(hw, hd) + margin
   509→      const dx = x - o.x, dz = z - o.z
   510→      if (dx * dx + dz * dz < reach * reach) return true
   511→    } else {
   512→      // axis-aligned box check (original behavior)
   513→      if (Math.abs(x - o.x) < hw + margin && Math.abs(z - o.z) < hd + margin) return true
   514→    }
   515→  }
   516→  return false
   517→}

   519→function pickSpawn(room: Room, preferTeam: Team = 'none'): [number, number, number] {
   520→  const map = getMap(room.mapId)
   521→  const spawns = map.spawns
   522→  // try each spawn with random offset, validate not inside obstacle
   523→  const trySpawn = (s: [number, number]): [number, number, number] => {
   524→    for (let attempt = 0; attempt < 8; attempt++) {
   525→      const x = s[0] + (Math.random() - 0.5) * 4
   526→      const z = s[1] + (Math.random() - 0.5) * 4
   527→      if (!spawnBlocked(map, x, z)) return [x, 1.7, z]
   528→    }
   529→    // fallback: use exact spawn point
   530→    return [s[0], 1.7, s[1]]
   531→  }
   532→  // PvP: bias by team (blue near spawn[0], red near spawn[1])
   533→  if (preferTeam !== 'none' && spawns.length >= 2) {
   534→    const idx = preferTeam === 'blue' ? 0 : 1
   535→    return trySpawn(spawns[idx])
   536→  }
   537→  // pick spawn farthest from any alive player
   538→  let best = spawns[0], bestD = -1
   539→  for (const s of spawns) {
   540→    let minD = Infinity
   541→    for (const p of room.players.values()) {
   542→      if (p.state !== 'alive') continue
   543→      minD = Math.min(minD, dist2D([s[0], 0, s[1]], p.pos))
   544→    }
   545→    if (minD > bestD) { bestD = minD; best = s }
   546→  }
   547→  return trySpawn(best)
   548→}

   723→function checkPvERoundEnd(r: Room) {
   724→  if (!r.roundActive) return
   725→  const aliveMobs = r.mobs.filter(m => m.state === 'alive').length
   726→  if (aliveMobs === 0) {
   727→    r.roundActive = false
   728→    r.roundEndsAt = Date.now() + ROUND_END_MS
   729→    r.nextLevel = r.level + 1
   730→    io.to(r.id).emit('room:mapChange', { mapId: r.mapId, level: r.level, mode: r.mode, banner: `¡Nivel ${r.level} superado!` })
   731→  }

   734→function checkPvPRoundEnd(r: Room) {
   735→  if (!r.roundActive || r.players.size < 2) return
   736→  const alivePlayers = Array.from(r.players.values()).filter(p => p.state === 'alive')
   737→  if (r.mode === 'ffa') {
   738→    // FFA: last man standing — round ends when ≤1 player is alive
   739→    if (alivePlayers.length <= 1) {
   740→      r.roundActive = false
   741→      r.roundEndsAt = Date.now() + ROUND_END_MS
   742→      const winnerName = alivePlayers[0]?.name ?? '—'
   743→      io.to(r.id).emit('room:mapChange', { mapId: r.mapId, level: r.level, mode: r.mode, banner: `¡${winnerName} gana la ronda!` })
   744→    }
   745→  } else {
   746→    // Team modes (1v1, 2v2, team): round ends when one team has no alive players
   747→    const teams = { blue: 0, red: 0 }
   748→    for (const p of alivePlayers) {
   749→      if (p.team === 'blue' || p.team === 'red') teams[p.team]++
   750→    }
   751→    if (teams.blue === 0 || teams.red === 0) {
   752→      r.roundActive = false
   753→      r.roundEndsAt = Date.now() + ROUND_END_MS
   754→      const winner = teams.blue > 0 ? 'Azul' : 'Rojo'
   755→      io.to(r.id).emit('room:mapChange', { mapId: r.mapId, level: r.level, mode: r.mode, banner: `¡Equipo ${winner} gana la ronda!` })
   756→    }
   757→  }
   758→}

   969→function applyDamageToPlayer(room: Room, target: Player, dmg: number, killerId: string, weapon: string, headshot: boolean, attackerPos?: [number, number, number]) {
   970→  const now = Date.now()
   971→  target.lastDamagedAt = now
   972→  // shield absorbs first
   973→  let remaining = dmg
   974→  if (target.shield > 0) {
   975→    const absorbed = Math.min(target.shield, remaining * 0.6)
   976→    target.shield -= absorbed
   977→    remaining -= absorbed
   978→  }
   979→  target.health -= remaining
   980→  io.to(room.id).emit('player:damaged', {
   981→    id: target.id, health: Math.max(0, target.health),
   982→    shield: Math.max(0, target.shield), by: killerId, headshot,
   983→    attackerPos: attackerPos ?? null,
   984→  })
   985→  if (target.health <= 0) killPlayer(room, target, killerId, weapon, headshot)
   986→}

  1027→function killPlayer(room: Room, victim: Player, killerId: string, weapon: string, headshot: boolean) {
  1028→  victim.state = 'dead'
  1029→  victim.health = 0
  1030→  victim.deaths += 1
  1031→  victim.streak = 0
  1032→  victim.respawnAt = Date.now() + RESPAWN_MS
  1033→  const killer = room.players.get(killerId)
  1034→  if (killer && killer.id !== victim.id) {
  1035→    killer.kills += 1
  1036→    killer.streak += 1
  1037→    killer.bestStreak = Math.max(killer.bestStreak, killer.streak)
  1038→    killer.score += headshot ? 150 : 100
  1039→    // streak rewards
  1040→    const reward = streakAt(killer.streak)
  1041→    if (reward && killer.streak === reward.streak) {
  1042→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
  1043→      io.to(killer.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
  1044→      applyStreakReward(room, killer, reward.id)
  1045→    } else {
  1046→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward: null })
  1047→    }
  1048→    // PvP modes: drop a random item on death
  1049→    if (isPvPMode(room.mode) && Math.random() < 0.6) dropItemOnMobKill(room, victim.pos)
  1050→  }
  1051→  io.to(room.id).emit('player:killed', { victimId: victim.id, killerId, weapon, headshot })
  1052→  killFeed(room, victim.name, killer?.name ?? null, weapon, headshot, 'player')
  1053→  setTimeout(() => {
  1054→    if (!victim.connected) return
  1055→    if (!room.players.has(victim.id)) return
  1056→    // Always respawn — round restarts will override with fresh state if needed
  1057→    victim.state = 'alive'
  1058→    victim.health = PLAYER_MAX_HP
  1059→    victim.shield = 0
  1060→    victim.pos = pickSpawn(room, victim.team)
  1061→    victim.weapon = 'pistol'
  1062→    victim.ammo = WEAPONS.pistol.magazine
  1063→    victim.reloading = false
  1064→    io.to(room.id).emit('player:respawned', { id: victim.id, pos: victim.pos })
  1065→  }, RESPAWN_MS)
  1066→  // round-end checks (PvP modes only — PvE round end is triggered by mob kills)
  1067→  if (isPvPMode(room.mode)) checkPvPRoundEnd(room)
  1068→}

  1070→function applyStreakReward(room: Room, p: Player, rewardId: string) {
  1071→  if (rewardId === 'heal' || rewardId === 'aura') {
  1072→    p.health = Math.min(PLAYER_MAX_HP, p.health + 25)
  1073→    p.shield = Math.min(PLAYER_MAX_SHIELD, p.shield + 25)
  1074→  }
  1075→  if (rewardId === 'bomb') {
  1076→    // damage all mobs near player + nearby enemies
  1077→    for (const m of room.mobs) {
  1078→      if (m.state !== 'alive') continue
  1079→      if (dist2D(m.pos, p.pos) < 12) {
  1080→        m.health -= 80
  1081→        m.hitFlash = Date.now()
  1082→        io.to(room.id).emit('mob:damaged', { id: m.id, health: Math.max(0, m.health), by: p.id })
  1083→        if (m.health <= 0) killMob(room, m, p.id, 'bomb', false)
  1084→      }
  1085→    }
  1086→    for (const other of room.players.values()) {
  1087→      if (other.id === p.id || other.state !== 'alive') continue
  1088→      // coop: no player damage; ffa/mixed: friendly fire on; team modes: skip same team
  1089→      if (room.mode === 'coop') continue
  1090→      if (!hasFriendlyFire(room.mode) && other.team === p.team && p.team !== 'none') continue
  1091→      if (dist2D(other.pos, p.pos) < 10) {
  1092→        applyDamageToPlayer(room, other, 60, p.id, 'bomb', false)
  1093→      }
  1094→    }
  1095→  }
  1096→  if (rewardId === 'drone') {
  1097→    room.droneOwner = p.id
  1098→    room.droneEnd = Date.now() + 20000
  1099→    io.to(room.id).emit('drone:state', { ownerId: p.id, pos: [p.pos[0], 3.5, p.pos[2]], expires: room.droneEnd, targetYaw: p.yaw })
  1100→  }
  1101→  if (rewardId === 'aura') {
  1102→    p.ammo = WEAPONS[p.weapon].magazine
  1103→    p.reloading = false
  1104→    io.to(p.id).emit('player:reloadDone', { weapon: p.weapon, ammo: p.ammo })
  1105→  }
  1106→}

  1108→function killMob(room: Room, mob: Mob, killerId: string, weapon: string, headshot: boolean) {
  1109→  mob.state = 'dead'
  1110→  mob.health = 0
  1111→  mob.respawnAt = Date.now() + MOB_RESPAWN_MS
  1112→  const killer = room.players.get(killerId)
  1113→  if (killer) {
  1114→    killer.streak += 1
  1115→    killer.bestStreak = Math.max(killer.bestStreak, killer.streak)
  1116→    killer.score += headshot ? 60 : 40
  1117→    const reward = streakAt(killer.streak)
  1118→    if (reward && killer.streak === reward.streak) {
  1119→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
  1120→      io.to(killer.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
  1121→      applyStreakReward(room, killer, reward.id)
  1122→    } else {
  1123→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward: null })
  1124→    }
  1125→  }
  1126→  io.to(room.id).emit('mob:killed', { id: mob.id, killerId, weapon, headshot })
  1127→  killFeed(room, 'Doodle Mob', killer?.name ?? null, weapon, headshot, 'mob')
  1128→  // item drop
  1129→  dropItemOnMobKill(room, mob.pos)
  1130→  checkPvERoundEnd(room)
  1131→  // schedule respawn
  1132→  setTimeout(() => {
  1133→    if (!rooms.has(room.id)) return
  1134→    if (!isPvEMode(room.mode)) return // mobs only respawn in PvE modes (coop/mixed) between levels
  1135→    mob.state = 'alive'
  1136→    mob.health = mob.maxHealth
  1137→    const a = Math.random() * Math.PI * 2
  1138→    const r = Math.random() * (HALF - 8)
  1139→    mob.pos = [Math.cos(a) * r, 1, Math.sin(a) * r]
  1140→    mob.wanderTarget = mob.pos.slice() as [number, number, number]
  1141→    io.to(room.id).emit('mob:respawned', { id: mob.id, pos: mob.pos })
  1142→  }, MOB_RESPAWN_MS)
  1143→}

  1145→// ----------------------- World tick -----------------------
  1146→setInterval(() => {
  1147→  const now = Date.now()
  1148→  // cleanup empty rooms
  1149→  for (const [rid, r] of rooms) {
  1150→    if (r.players.size === 0) {
  1151→      if (r.emptySince == null) r.emptySince = now
  1152→      if (now - (r.emptySince as number) > EMPTY_ROOM_TTL_MS) {
  1153→        // keep at least the default PvE room
  1154→        if (rooms.size > 1) {
  1155→          rooms.delete(rid)
  1156→          continue
  1157→        }
  1158→      }
  1159→    }
  1160→  }
  1161→
  1162→  for (const room of rooms.values()) {
  1163→    if (room.players.size === 0) continue
  1164→
  1165→    // round transitions
  1166→    if (!room.roundActive && room.roundEndsAt && now >= room.roundEndsAt) {
  1167→      room.roundEndsAt = 0
  1168→      if (isPvEMode(room.mode)) startPvERound(room, room.nextLevel)
  1169→      else startPvPRound(room, true)
  1170→    }
  1171→
  1172→    // mob AI (PvE modes: coop + mixed)
  1173→    if (isPvEMode(room.mode)) {
  1174→      const cfg = pveLevelConfig(room.level)
  1175→      for (const mob of room.mobs) {
  1176→        if (mob.state === 'dead') continue
  1177→        let nearest: Player | null = null
  1178→        let nd = Infinity
  1179→        for (const p of room.players.values()) {
  1180→          if (p.state !== 'alive') continue
  1181→          const d = dist2D(mob.pos, p.pos)
  1182→          if (d < nd) { nd = d; nearest = p }
  1183→        }
  1184→        let tx: number, tz: number
  1185→        if (nearest && nd < 16) {
  1186→          tx = nearest.pos[0]; tz = nearest.pos[2]
  1187→          if (nd < 1.9 && now - mob.lastAttack > MOB_ATTACK_CD) {
  1188→            mob.lastAttack = now
  1189→            applyDamageToPlayer(room, nearest, cfg.mobDamage, mob.id, 'mob', false, mob.pos)
  1190→          }
  1191→        } else {
  1192→          if (dist2D(mob.pos, mob.wanderTarget) < 1.5) {
  1193→            const a = Math.random() * Math.PI * 2
  1194→            const r = Math.random() * (HALF - 6)
  1195→            mob.wanderTarget = [Math.cos(a) * r, 1, Math.sin(a) * r]
  1196→          }
  1197→          tx = mob.wanderTarget[0]; tz = mob.wanderTarget[2]
  1198→        }
  1199→        const dx = tx - mob.pos[0], dz = tz - mob.pos[2]
  1200→        const len = Math.hypot(dx, dz) || 1
  1201→        const vx = (dx / len) * cfg.mobSpeed, vz = (dz / len) * cfg.mobSpeed
  1202→        mob.vel[0] = vx; mob.vel[1] = vz
  1203→        const step = TICK_MS / 1000
  1204→        mob.pos[0] += vx * step
  1205→        mob.pos[2] += vz * step
  1206→        clampToArena(mob.pos)
  1207→        mob.pos[1] = 1
  1208→      }
  1209→      io.to(room.id).emit('mob:state', { mobs: room.mobs.map(m => ({ id: m.id, pos: m.pos, state: m.state, flash: m.hitFlash })) })
  1210→    }
  1211→
  1212→    // PvE health regen: if player hasn't taken damage for 5s, regen 1.5 HP/sec
  1213→    if (isPvEMode(room.mode)) {
  1214→      for (const p of room.players.values()) {
  1215→        if (p.state !== 'alive') continue
  1216→        if (p.health >= PLAYER_MAX_HP) continue
  1217→        if (now - p.lastDamagedAt > 5000) {
  1218→          p.health = Math.min(PLAYER_MAX_HP, p.health + 1.5 * (TICK_MS / 1000) * 4)
  1219→          // send updated health to the player (throttled)
  1220→          if (now % 500 < TICK_MS) {
  1221→            io.to(p.id).emit('player:damaged', { id: p.id, health: Math.round(p.health), shield: p.shield, by: 'regen', headshot: false, attackerPos: null, regen: true })
  1222→          }
  1223→        }
  1224→      }
  1225→    }
  1226→
  1227→    // drone streak reward: auto-attack nearest enemy/mob
  1228→    if (room.droneOwner && now < room.droneEnd) {
  1229→      const owner = room.players.get(room.droneOwner)
  1230→      if (owner && owner.state === 'alive') {
  1231→        // attack nearest mob every ~700ms
  1232→        if (now % 700 < TICK_MS) {
  1233→          let target: Mob | null = null; let td = Infinity
  1234→          for (const m of room.mobs) {
  1235→            if (m.state !== 'alive') continue
  1236→            const d = dist2D(m.pos, owner.pos)
  1237→            if (d < 18 && d < td) { td = d; target = m }
  1238→          }
  1239→          if (target) {
  1240→            target.health -= 14
  1241→            target.hitFlash = now
  1242→            io.to(room.id).emit('mob:damaged', { id: target.id, health: Math.max(0, target.health), by: owner.id })
  1243→            if (target.health <= 0) killMob(room, target, owner.id, 'drone', false)
  1244→          }
  1245→        }
  1246→      } else if (owner && owner.state !== 'alive') {
  1247→        // drone persists but does nothing while dead
  1248→      } else {
  1249→        room.droneOwner = null
  1250→      }
  1251→    } else if (room.droneOwner && now >= room.droneEnd) {
  1252→      room.droneOwner = null
  1253→      io.to(room.id).emit('drone:state', { ownerId: null, pos: null, expires: 0 })
  1254→    }
  1255→    // emit drone position to all clients (for 3D rendering)
  1256→    if (room.droneOwner && now < room.droneEnd) {
  1257→      const owner = room.players.get(room.droneOwner)
  1258→      if (owner && owner.state === 'alive') {
  1259→        // drone hovers above and behind the owner
  1260→        const dx = -Math.sin(owner.yaw) * 3
  1261→        const dz = -Math.cos(owner.yaw) * 3
  1262→        const dronePos: [number, number, number] = [owner.pos[0] + dx, 3.5, owner.pos[2] + dz]
  1263→        io.to(room.id).emit('drone:state', { ownerId: room.droneOwner, pos: dronePos, expires: room.droneEnd, targetYaw: owner.yaw })
  1264→      }
  1265→    }
  1266→
  1267→    // item expiry
  1268→    let itemsChanged = false
  1269→    room.items = room.items.filter(it => {
  1270→      if (now - it.born > ITEM_LIFETIME_MS) { itemsChanged = true; return false }
  1271→      return true
  1272→    })
  1273→    if (itemsChanged) {
  1274→      io.to(room.id).emit('items:state', { items: room.items.map(i => ({ id: i.id, type: i.type, pos: i.pos })) })
  1275→    }
  1276→
  1277→    // periodic item spawns (PvE: every ~12s, PvP: every ~16s)
  1278→    room.itemSpawnTimer += TICK_MS
  1279→    const interval = isPvEMode(room.mode) ? ITEM_RESPAWN_INTERVAL_MS : 16000
  1280→    if (room.itemSpawnTimer >= interval && room.items.length < 6) {

  1286→
  1287→    // roster every ~1s
  1288→    if (now % 1000 < TICK_MS) emitRoomState(room)
