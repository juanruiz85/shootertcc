    56→    56→function buildHouse(cx: number, cz: number, w: number, d: number, h: number, color: number, roofColor: number, doorSide: 'N'|'S'|'E'|'W' = 'S'): MapObstacle[] {
    57→    57→  const walls: MapObstacle[] = []
    58→    58→  const wallT = 0.3
    59→    59→  const doorW = 1.5
    60→    60→  const winH = 1.0, winY = 1.2
    61→    61→  if (doorSide !== 'N') { walls.push(ob(cx, cz - d/2, w, h, wallT, false, color, 'wall')) }
    62→    62→  else { walls.push(ob(cx - w/4 - doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx + w/4 + doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx, cz - d/2, doorW, h - winY - 0.5, wallT, false, color, 'wall', 0, true)) }
    63→    63→  if (doorSide !== 'S') { walls.push(ob(cx, cz + d/2, w, h, wallT, false, color, 'wall')) }
    64→    64→  else { walls.push(ob(cx - w/4 - doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx + w/4 + doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx, cz + d/2, doorW, h - winY - 0.5, wallT, false, color, 'wall', 0, true)) }
    65→    65→  if (doorSide !== 'E') { walls.push(ob(cx + w/2, cz, wallT, h, d, false, color, 'wall')) }
    66→    66→  else { walls.push(ob(cx + w/2, cz - d/4 - doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')); walls.push(ob(cx + w/2, cz + d/4 + doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')) }

    73→    73→function buildStairs(cx: number, cz: number, steps: number, dir: 'N'|'S'|'E'|'W', color: number): MapObstacle[] {
    74→    74→  const result: MapObstacle[] = []
    75→    75→  const stepH = 0.5, stepD = 0.6, stepW = 2
    76→    76→  for (let i = 0; i < steps; i++) { let x = cx, z = cz; if (dir === 'N') z = cz - i * stepD; if (dir === 'S') z = cz + i * stepD; if (dir === 'E') x = cx + i * stepD; if (dir === 'W') x = cx - i * stepD; result.push(ob(x, z, stepW, stepH + 0.01, stepD, true, color, 'stair')) }
    77→    77→  return result
    78→    78→}

   450→   450→function spawnBlocked(map: GameMap, x: number, z: number): boolean {
   451→   451→  // check if a position is inside any obstacle (with margin)
   452→   452→  // skip water and noCollide obstacles (they don't block spawning)
   453→   453→  const margin = 1.5
   454→   454→  for (const o of map.obstacles) {
   455→   455→    if (o.kind === 'water' || o.noCollide) continue
   456→   456→    const hw = o.w / 2, hd = o.d / 2
   457→   457→    if (o.rotation && o.rotation !== 0) {
   458→   458→      // rotated: use slightly larger margin (bounding-circle approximation)
   459→   459→      const reach = Math.max(hw, hd) + margin
   460→   460→      const dx = x - o.x, dz = z - o.z
   461→   461→      if (dx * dx + dz * dz < reach * reach) return true
   462→   462→    } else {
   463→   463→      // axis-aligned box check (original behavior)
   464→   464→      if (Math.abs(x - o.x) < hw + margin && Math.abs(z - o.z) < hd + margin) return true
   465→   465→    }
   466→   466→  }
   467→   467→  return false
   468→   468→}

   470→   470→function pickSpawn(room: Room, preferTeam: Team = 'none'): [number, number, number] {
   471→   471→  const map = getMap(room.mapId)
   472→   472→  const spawns = map.spawns
   473→   473→  // try each spawn with random offset, validate not inside obstacle
   474→   474→  const trySpawn = (s: [number, number]): [number, number, number] => {
   475→   475→    for (let attempt = 0; attempt < 8; attempt++) {
   476→   476→      const x = s[0] + (Math.random() - 0.5) * 4
   477→   477→      const z = s[1] + (Math.random() - 0.5) * 4
   478→   478→      if (!spawnBlocked(map, x, z)) return [x, 1.7, z]
   479→   479→    }
   480→   480→    // fallback: use exact spawn point
   481→   481→    return [s[0], 1.7, s[1]]
   482→   482→  }
   483→   483→  // PvP: bias by team (blue near spawn[0], red near spawn[1])
   484→   484→  if (preferTeam !== 'none' && spawns.length >= 2) {
   485→   485→    const idx = preferTeam === 'blue' ? 0 : 1
   486→   486→    return trySpawn(spawns[idx])
   487→   487→  }
   488→   488→  // pick spawn farthest from any alive player
   489→   489→  let best = spawns[0], bestD = -1
   490→   490→  for (const s of spawns) {
   491→   491→    let minD = Infinity
   492→   492→    for (const p of room.players.values()) {
   493→   493→      if (p.state !== 'alive') continue
   494→   494→      minD = Math.min(minD, dist2D([s[0], 0, s[1]], p.pos))
   495→   495→    }
   496→   496→    if (minD > bestD) { bestD = minD; best = s }
   497→   497→  }
   498→   498→  return trySpawn(best)
   499→   499→}

   501→   501→function makePlayer(socketId: string, name: string, skin: string, room: Room): Player {
   502→   502→  // team assignment: only team modes (1v1, 2v2, team) get blue/red; ffa/coop/mixed = 'none'
   503→   503→  let team: Team = 'none'
   504→   504→  if (room.mode === '1v1' || room.mode === '2v2' || room.mode === 'team') {
   505→   505→    const counts = { blue: 0, red: 0 }
   506→   506→    for (const p of room.players.values()) counts[p.team]++
   507→   507→    team = counts.blue <= counts.red ? 'blue' : 'red'
   508→   508→  }
   509→   509→  // ffa, coop, mixed → team stays 'none'

   685→   685→function checkPvPRoundEnd(r: Room) {
   686→   686→  if (!r.roundActive || r.players.size < 2) return
   687→   687→  const alivePlayers = Array.from(r.players.values()).filter(p => p.state === 'alive')
   688→   688→  if (r.mode === 'ffa') {
   689→   689→    // FFA: last man standing — round ends when ≤1 player is alive
   690→   690→    if (alivePlayers.length <= 1) {
   691→   691→      r.roundActive = false
   692→   692→      r.roundEndsAt = Date.now() + ROUND_END_MS
   693→   693→      const winnerName = alivePlayers[0]?.name ?? '—'
   694→   694→      io.to(r.id).emit('room:mapChange', { mapId: r.mapId, level: r.level, mode: r.mode, banner: `¡${winnerName} gana la ronda!` })
   695→   695→    }
   696→   696→  } else {
   697→   697→    // Team modes (1v1, 2v2, team): round ends when one team has no alive players
   698→   698→    const teams = { blue: 0, red: 0 }
   699→   699→    for (const p of alivePlayers) {
   700→   700→      if (p.team === 'blue' || p.team === 'red') teams[p.team]++
   701→   701→    }
   702→   702→    if (teams.blue === 0 || teams.red === 0) {
   703→   703→      r.roundActive = false
   704→   704→      r.roundEndsAt = Date.now() + ROUND_END_MS
   705→   705→      const winner = teams.blue > 0 ? 'Azul' : 'Rojo'
   706→   706→      io.to(r.id).emit('room:mapChange', { mapId: r.mapId, level: r.level, mode: r.mode, banner: `¡Equipo ${winner} gana la ronda!` })
   707→   707→    }
   708→   708→  }

   920→   920→function applyDamageToPlayer(room: Room, target: Player, dmg: number, killerId: string, weapon: string, headshot: boolean, attackerPos?: [number, number, number]) {

   927→   927→    target.shield -= absorbed

   929→   929→  }

   932→   932→    id: target.id, health: Math.max(0, target.health),
   933→   933→    shield: Math.max(0, target.shield), by: killerId, headshot,
   934→   934→    attackerPos: attackerPos ?? null,

   978→   978→function killPlayer(room: Room, victim: Player, killerId: string, weapon: string, headshot: boolean) {
   979→   979→  victim.state = 'dead'
   980→   980→  victim.health = 0
   981→   981→  victim.deaths += 1
   982→   982→  victim.streak = 0
   983→   983→  victim.respawnAt = Date.now() + RESPAWN_MS
   984→   984→  const killer = room.players.get(killerId)
   985→   985→  if (killer && killer.id !== victim.id) {
   986→   986→    killer.kills += 1
   987→   987→    killer.streak += 1
   988→   988→    killer.bestStreak = Math.max(killer.bestStreak, killer.streak)
   989→   989→    killer.score += headshot ? 150 : 100
   990→   990→    // streak rewards
   991→   991→    const reward = streakAt(killer.streak)
   992→   992→    if (reward && killer.streak === reward.streak) {
   993→   993→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
   994→   994→      io.to(killer.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
   995→   995→      applyStreakReward(room, killer, reward.id)
   996→   996→    } else {
   997→   997→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward: null })
   998→   998→    }
   999→   999→    // PvP modes: drop a random item on death
  1000→  1000→    if (isPvPMode(room.mode) && Math.random() < 0.6) dropItemOnMobKill(room, victim.pos)
  1001→  1001→  }
  1002→  1002→  io.to(room.id).emit('player:killed', { victimId: victim.id, killerId, weapon, headshot })
  1003→  1003→  killFeed(room, victim.name, killer?.name ?? null, weapon, headshot, 'player')
  1004→  1004→  setTimeout(() => {
  1005→  1005→    if (!victim.connected) return
  1006→  1006→    if (!room.players.has(victim.id)) return
  1007→  1007→    // Always respawn — round restarts will override with fresh state if needed
  1008→  1008→    victim.state = 'alive'
  1009→  1009→    victim.health = PLAYER_MAX_HP
  1010→  1010→    victim.shield = 0
  1011→  1011→    victim.pos = pickSpawn(room, victim.team)
  1012→  1012→    victim.weapon = 'pistol'
  1013→  1013→    victim.ammo = WEAPONS.pistol.magazine
  1014→  1014→    victim.reloading = false
  1015→  1015→    io.to(room.id).emit('player:respawned', { id: victim.id, pos: victim.pos })
  1016→  1016→  }, RESPAWN_MS)
  1017→  1017→  // round-end checks (PvP modes only — PvE round end is triggered by mob kills)
  1018→  1018→  if (isPvPMode(room.mode)) checkPvPRoundEnd(room)
  1019→  1019→}

  1021→  1021→function applyStreakReward(room: Room, p: Player, rewardId: string) {
  1022→  1022→  if (rewardId === 'heal' || rewardId === 'aura') {
  1023→  1023→    p.health = Math.min(PLAYER_MAX_HP, p.health + 25)
  1024→  1024→    p.shield = Math.min(PLAYER_MAX_SHIELD, p.shield + 25)
  1025→  1025→  }
  1026→  1026→  if (rewardId === 'bomb') {
  1027→  1027→    // damage all mobs near player + nearby enemies
  1028→  1028→    for (const m of room.mobs) {
  1029→  1029→      if (m.state !== 'alive') continue
  1030→  1030→      if (dist2D(m.pos, p.pos) < 12) {
  1031→  1031→        m.health -= 80
  1032→  1032→        m.hitFlash = Date.now()
  1033→  1033→        io.to(room.id).emit('mob:damaged', { id: m.id, health: Math.max(0, m.health), by: p.id })
  1034→  1034→        if (m.health <= 0) killMob(room, m, p.id, 'bomb', false)
  1035→  1035→      }
  1036→  1036→    }
  1037→  1037→    for (const other of room.players.values()) {
  1038→  1038→      if (other.id === p.id || other.state !== 'alive') continue
  1039→  1039→      // coop: no player damage; ffa/mixed: friendly fire on; team modes: skip same team
  1040→  1040→      if (room.mode === 'coop') continue
  1041→  1041→      if (!hasFriendlyFire(room.mode) && other.team === p.team && p.team !== 'none') continue
  1042→  1042→      if (dist2D(other.pos, p.pos) < 10) {
  1043→  1043→        applyDamageToPlayer(room, other, 60, p.id, 'bomb', false)
  1044→  1044→      }
  1045→  1045→    }
  1046→  1046→  }
  1047→  1047→  if (rewardId === 'drone') {
  1048→  1048→    room.droneOwner = p.id
  1049→  1049→    room.droneEnd = Date.now() + 20000
  1050→  1050→    io.to(room.id).emit('drone:state', { ownerId: p.id, pos: [p.pos[0], 3.5, p.pos[2]], expires: room.droneEnd, targetYaw: p.yaw })
  1051→  1051→  }
  1052→  1052→  if (rewardId === 'aura') {
  1053→  1053→    p.ammo = WEAPONS[p.weapon].magazine

  1055→  1055→    io.to(p.id).emit('player:reloadDone', { weapon: p.weapon, ammo: p.ammo })
  1056→  1056→  }
