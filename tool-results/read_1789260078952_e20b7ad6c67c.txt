   202→function doodle(geo, color, thickness = 1.015) {
   203→    const g = new THREE.Group();
   204→    const mat = new THREE.MeshToonMaterial({ color });
   205→    const mesh = new THREE.Mesh(geo, mat);
   206→    const edges = new THREE.EdgesGeometry(geo, 20);
   207→    const wire = new THREE.LineSegments(edges, new THREE.LineBasicMaterial({ color: 0x000000, linewidth: 2 }));
   208→    wire.scale.setScalar(thickness);
   209→    g.add(mesh, wire);
   210→    g.userData.isDoodle = true;
   211→    g.userData.wireframe = wire;
   212→    g.userData.wobbleSpeed = 0.003 + Math.random() * 0.002;
   213→    g.userData.wobbleOffset = Math.random() * 100;
   214→    return g;
   215→}

   339→// ============================================================
   340→// GENERADOR DE MAPAS
   341→// ============================================================
   342→function buildMap(mapId) {
   343→    while (mapGroup.children.length) mapGroup.remove(mapGroup.children[0]);
   344→    state.obstacles = [];
   345→    
   346→    const map = MAPS.find(m => m.id === mapId);
   347→    
   348→    // Suelo
   349→    const ground = doodle(new THREE.PlaneGeometry(100, 100, 20, 20), map.ground);
   350→    ground.rotation.x = -Math.PI/2;
   351→    // Ligera deformación doodle
   352→    const pos = ground.children[0].geometry.attributes.position;
   353→    for (let i = 0; i < pos.count; i++) {
   354→        pos.setZ(i, pos.getZ(i) + (Math.random() - 0.5) * 0.15);
   355→    }
   356→    pos.needsUpdate = true;
   357→    ground.children[0].geometry.computeVertexNormals();
   358→    mapGroup.add(ground);
   359→    
   360→    // Paredes perimetrales (colisión)
   361→    const wallH = 8;
   362→    for (let i = 0; i < 4; i++) {
   363→        const wall = doodle(new THREE.BoxGeometry(100, wallH, 1), map.walls);
   364→        wall.position.y = wallH/2;
   365→        if (i === 0) wall.position.z = -50;
   366→        if (i === 1) wall.position.z = 50;
   367→        if (i === 2) { wall.position.x = -50; wall.rotation.y = Math.PI/2; }
   368→        if (i === 3) { wall.position.x = 50; wall.rotation.y = Math.PI/2; }
   369→        mapGroup.add(wall);
   370→        state.obstacles.push({ mesh: wall, box: new THREE.Box3().setFromObject(wall) });
   371→    }
   372→    
   373→    // Generar obstáculos según tema
   374→    const theme = map.theme;
   375→    const obstacleCount = 35 + Math.floor(Math.random() * 15);
   376→    
   377→    for (let i = 0; i < obstacleCount; i++) {
   378→        let w, h, d, color;
   379→        
   380→        // Variar según tema
   381→        if (theme === 'indoor' || theme === 'industrial') {
   382→            w = 2 + Math.random() * 3; h = 3 + Math.random() * 4; d = 0.3 + Math.random() * 0.5;
   383→            color = Math.random() > 0.5 ? map.walls : map.accent;
   384→        } else if (theme === 'natural') {
   385→            w = 1 + Math.random() * 2; h = 2 + Math.random() * 3; d = 1 + Math.random() * 2;
   386→            color = Math.random() > 0.6 ? map.accent : map.walls;
   387→        } else if (theme === 'urban') {
   388→            w = 3 + Math.random() * 5; h = 4 + Math.random() * 8; d = 3 + Math.random() * 5;
   389→            color = Math.random() > 0.3 ? map.walls : map.accent;
   390→        } else if (theme === 'open') {
   391→            w = 1 + Math.random() * 3; h = 1 + Math.random() * 2; d = 1 + Math.random() * 3;
   392→            color = map.accent;
   393→        } else { // sci-fi, extreme
   394→            w = 2 + Math.random() * 3; h = 3 + Math.random() * 5; d = 2 + Math.random() * 3;
   395→            color = map.accent;

   461→function checkCollision(newPos) {
   462→    const playerBox = new THREE.Box3().setFromCenterAndSize(
   463→        newPos, new THREE.Vector3(0.6, 1.7, 0.6)
   464→    );
   465→    
   466→    for (const obs of state.obstacles) {
   467→        if (playerBox.intersectsBox(obs.box)) return true;
   468→    }
   469→    
   470→    // Límites del mapa
   471→    if (Math.abs(newPos.x) > 48 || Math.abs(newPos.z) > 48) return true;
   472→    
   473→    return false;
   474→}

   476→function checkGround(pos) {
   477→    // Verificar si hay suelo debajo debajo del jugador
   478→    const rayOrigin = new THREE.Vector3(pos.x, pos.y + 2, pos.z);
   479→    const rayDir = new THREE.Vector3(0, -1, 0);
   480→    const ray = new THREE.Raycaster(rayOrigin, rayDir, 0, 4);
   481→    
   482→    for (const obs of state.obstacles) {
   483→        const hits = ray.intersectObject(obs.mesh, true);
   484→        if (hits.length > 0) {
   485→            const top = hits[0].point.y;
   486→            return top;
   487→        }
   488→    }
   489→    return 0; // suelo base
   490→}

   495→class Enemy {
   496→    constructor(level) {

   498→        
   499→        // Cuerpo (más principal - daño de 25 daño)
   500→        this.body = doodle(new THREE.BoxGeometry(0.7, 1.2, 0.5), 0xff6b6b);
   501→        this.body.position.y = 0.6;
   502→        this.group.add(this.body);
   503→        
   504→        // Cabeza (hitbox pequeña - daño x3)
   505→        this.head = doodle(new THREE.BoxGeometry(0.4, 0.4, 0.4), 0xffdbac);
   506→        this.head.position.y = 1.4;
   507→        this.body.add(this.head);
   508→        
   509→        // Brazos (decoración)
   510→        const armL = doodle(new THREE.BoxGeometry(0.15, 0.7, 0.15), 0xff6b6b);
   511→        armL.position.set(-0.35, 0.3, 0);
   512→        this.body.add(armL);
   513→        const armR = doodle(new THREE.BoxGeometry(0.15, 0.7, 0.15), 0xff6b6b);
   514→        armR.position.set(0.35, 0.3, 0);
   515→        this.body.add(armR);
   516→        
   517→        // Posición inicial
   518→        const angle = Math.random() * Math.PI * 2;
   519→        const dist = 20 + Math.random() * 20;
   520→        this.group.position.set(Math.cos(angle) * dist, 0, Math.sin(angle) * dist);
   521→        
   522→        this.health = 50 + level * 10;
   523→        this.maxHealth = this.health;
   524→        this.level = level;
   525→        this.lastShot = 0;
   526→        this.shootCooldown = Math.max(600, 2000 - level * 80);
   527→        
   528→        // El enemigo asignada según nivel
   529→        if (level <= 3) this.weapon = 'pistol';
   530→        else if (level <= 6) this.weapon = Math.random() > 0.5 ? 'smg' : 'pistol';
   531→        else if (level <= 10) this.weapon = ['smg', 'shotgun', 'rifle'][Math.floor(Math.random()*3)];
   532→        else if (level <= 15) this.weapon = ['rifle', 'mg', 'shotgun'][Math.floor(Math.random()*3)];
   533→        else this.weapon = ['mg', 'sniper', 'minigun'][Math.floor(Math.random()*3)];
   534→        
   535→        this.velocity = new THREE.Vector3();
   536→        this.targetPos = this.group.position.clone();
   537→        this.alive = true;
   538→        this.hitFlash = 0;
   539→        
   540→        scene.add(this.group);
   541→    }
   542→    
   543→    update(dt) {
   544→        if (!this.alive) return;
   545→        
   546→        // Mover hacia el jugador con IA básica
   547→        const dir = new THREE.Vector3().subVectors(state.player.pos, this.group.position);
   548→        dir.y = 0;
   549→        const dist = dir.length();
   550→        dir.normalize();
   551→        
   552→        // Mantener distancia de ataque
   553→        const idealDist = 12;
   554→        const speed = 2 + this.level * 0.15;
   555→        
   556→        if (dist > idealDist + 2) {
   557→            this.group.position.x += dir.x * speed * dt;
   558→            this.group.position.z += dir.z * speed * dt;
   559→        } else if (dist < idealDist - 2) {
   560→            this.group.position.x -= dir.x * speed * dt;
   561→            this.group.position.z -= dir.z * speed * dt;
   562→        }
   563→        
   564→        // Mirar al jugador
   565→        this.group.lookAt(state.player.pos.x, this.group.position.y, state.player.pos.z);
   566→        
   567→        // Disparar
   568→        const now = Date.now();
   569→        if (dist < 30 && now - this.lastShot > this.shootCooldown) {
   570→            this.shoot();
   571→            this.lastShot = now;
   572→        }
   573→        
   574→        // Ground check
   575→        const ground = checkGround(this.group.position);
   576→        this.group.position.y = ground;
   577→        
   578→        // Flash de daño
   579→        if (this.hitFlash > 0) {
   580→            this.body.children[0].material.color.setHex(0xffffff);
   581→            this.hitFlash -= dt;
   582→            if (this.hitFlash <= 0) this.body.children[0].material.color.setHex(0xff6b6b);
   583→        }
   584→        
   585→        // Animación doodle
   586→        this.body.userData.wireframe.rotation.z = Math.sin(Date.now() * 0.005 + this.group.id) * 0.03;
   587→    }
   588→    
   589→    shoot() {
   590→        const wp = WEAPONS[this.weapon];
   591→        const origin = this.group.position.clone();
   592→        origin.y += 1.2;
   593→        
   594→        for (let i = 0; i < Math.min(wp.pellets, 3); i++) {
   595→            const dir = new THREE.Vector3().subVectors(state.player.pos, origin).normalize();
   596→            // Añadir imprecisión
   597→            dir.x += (Math.random() - 0.5) * 0.15;
   598→            dir.y += (Math.random() - 0.5) * 0.08;
   599→            dir.z += (Math.random() - 0.5) * 0.15;
   600→            dir.normalize();
   601→            
   602→            const bullet = new THREE.Mesh(
   603→                new THREE.SphereGeometry(0.08, 6, 6),
   604→                new THREE.MeshBasicMaterial({ color: 0xff0000 })
   605→            );
   606→            bullet.position.copy(origin);
   607→            bullet.userData = { velocity: dir.multiplyScalar(wp.bulletSpeed), damage: wp.damage * 0.5, life: 100 };
   608→            scene.add(bullet);
   609→            state.enemyBullets.push(bullet);
   610→        }
   611→    }
   612→    
   613→    hit(damage, isHeadshot) {
   614→        const actualDmg = isHeadshot ? damage * 3 : damage;
   615→        this.health -= actualDmg;
   616→        this.hitFlash = 0.15;
   617→        
   618→        if (this.health <= 0) {
   619→            this.die();
   620→        }
   621→    }
   622→    
   623→    die() {
   624→        this.alive = false;
   625→        scene.remove(this.group);
   626→        
   627→        state.player.kills++;
   628→        state.player.streak++;
   629→        state.totalKills++;
   630→        addKillFeed(`💀 Enemigo eliminado (${isHeadshot ? 'HEADSHOT' : 'kill'})`);
   631→        document.getElementById('kills').innerText = state.player.kills;
   632→        document.getElementById('streak').innerText = state.player.streak;
   633→        
   634→        // Drop aleatorio (60% probabilidad)
   635→        if (Math.random() < 0.6) {
   636→            spawnDrop(this.group.position);
   637→        }
   638→        
   639→        // Probabilidad de drone en nivel >= 10
   640→        if (this.level >= 10 && Math.random() < 0.08) {
   641→            spawnDrone();
   642→        }
   643→        
   644→        // Killstreak: bomba a los 10 kills sin morir
   645→        if (state.player.streak === 10) {
   646→            state.player.bombReady = true;
   647→            document.getElementById('bombReady').classList.remove('hidden');
   648→            addKillFeed('💣 ¡BOMBA DISPONIBLE! [B]');
   649→        }
   650→        
   651→        // Progreso de nivel
   652→        checkLevelProgress();
   653→    }
   654→}

   797→        dir.y += (Math.random() - 0.5) * wp.spread;

   801→        bullet.userData = { 
   802→            velocity: dir.multiplyScalar(wp.bulletSpeed),
