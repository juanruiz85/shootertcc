    98→    98→// Build a house: 4 walls with a door gap on one side, windows, optional roof
    99→    99→// Returns array of obstacles (wall segments)
   100→   100→function buildHouse(cx: number, cz: number, w: number, d: number, h: number, color: number, roofColor: number, doorSide: 'N'|'S'|'E'|'W' = 'S'): MapObstacle[] {
   101→   101→  const walls: MapObstacle[] = []
   102→   102→  const wallT = 0.3
   103→   103→  const doorW = 1.5
   104→   104→  const doorH = 2.2  // door height
   105→   105→  const winH = 1.0
   106→   106→  // North wall (z = cz - d/2)
   107→   107→  if (doorSide !== 'N') {
   108→   108→    walls.push(ob(cx, cz - d/2, w, h, wallT, false, color, 'wall'))
   109→   109→  } else {
   110→   110→    walls.push(ob(cx - w/4 - doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall'))
   111→   111→    walls.push(ob(cx + w/4 + doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall'))
   112→   112→    // window above door (positioned at doorH, no collision)
   113→   113→    walls.push(ob(cx, cz - d/2, doorW, h - doorH, wallT, false, color, 'wall', 0, true, doorH))
   114→   114→  }
   115→   115→  // South wall (z = cz + d/2)
   116→   116→  if (doorSide !== 'S') {
   117→   117→    walls.push(ob(cx, cz + d/2, w, h, wallT, false, color, 'wall'))
   118→   118→  } else {
   119→   119→    walls.push(ob(cx - w/4 - doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall'))
   120→   120→    walls.push(ob(cx + w/4 + doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall'))
   121→   121→    walls.push(ob(cx, cz + d/2, doorW, h - doorH, wallT, false, color, 'wall', 0, true, doorH))
   122→   122→  }
   123→   123→  // East wall (x = cx + w/2)
   124→   124→  if (doorSide !== 'E') {
   125→   125→    walls.push(ob(cx + w/2, cz, wallT, h, d, false, color, 'wall'))
   126→   126→  } else {
   127→   127→    walls.push(ob(cx + w/2, cz - d/4 - doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall'))
   128→   128→    walls.push(ob(cx + w/2, cz + d/4 + doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall'))
   129→   129→  }
   130→   130→  // West wall (x = cx - w/2)
   131→   131→  if (doorSide !== 'W') {
   132→   132→    walls.push(ob(cx - w/2, cz, wallT, h, d, false, color, 'wall'))
   133→   133→  } else {
   134→   134→    walls.push(ob(cx - w/2, cz - d/4 - doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall'))
   135→   135→    walls.push(ob(cx - w/2, cz + d/4 + doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall'))
   136→   136→  }
   137→   137→  // Roof at top of walls (player can stand on it)
   138→   138→  walls.push(ob(cx, cz, w + 0.6, 0.3, d + 0.6, true, roofColor, 'roof', 0, false, h))
   139→   139→  return walls
   140→   140→}

   142→   142→// Build stairs: a series of steps going up, each at increasing height
   143→   143→function buildStairs(cx: number, cz: number, steps: number, dir: 'N'|'S'|'E'|'W', color: number): MapObstacle[] {
   144→   144→  const result: MapObstacle[] = []
   145→   145→  const stepH = 0.6, stepD = 0.7, stepW = 2.5
   146→   146→  for (let i = 0; i < steps; i++) {
   147→   147→    let x = cx, z = cz
   148→   148→    if (dir === 'N') z = cz - i * stepD
   149→   149→    if (dir === 'S') z = cz + i * stepD
   150→   150→    if (dir === 'E') x = cx + i * stepD
   151→   151→    if (dir === 'W') x = cx - i * stepD
   152→   152→    // each step is positioned at y = i * stepH (stacking upward)
   153→   153→    result.push(ob(x, z, stepW, stepH, stepD, true, color, 'stair', 0, false, i * stepH))
   154→   154→  }
   155→   155→  return result
   156→   156→}

   158→   158→// Build a tree (trunk + foliage on top)
   159→   159→function buildTree(cx: number, cz: number, scale: number = 1): MapObstacle[] {
   160→   160→  const trunkH = 3 * scale
   161→   161→  const foliageH = 2.5 * scale
   162→   162→  return [
   163→   163→    // trunk at ground level
   164→   164→    ob(cx, cz, 0.4 * scale, trunkH, 0.4 * scale, false, 0x7a5230, 'cyl'),
   165→   165→    // foliage on top of trunk
   166→   166→    ob(cx, cz, 2.5 * scale, foliageH, 2.5 * scale, true, 0x27ae60, 'box', 0, false, trunkH),
   167→   167→  ]
   168→   168→}

   170→   170→// Build a car (body + cabin on top)
   171→   171→function buildCar(cx: number, cz: number, color: number, rotation: number = 0): MapObstacle[] {
   172→   172→  return [
   173→   173→    ob(cx, cz, 2, 0.8, 4, true, color, 'box', rotation),           // body at ground
   174→   174→    ob(cx, cz, 1.8, 0.6, 2, false, 0x2c3e50, 'box', rotation, false, 0.8), // cabin on top
   175→   175→  ]
   176→   176→}

   178→   178→// Build a multi-floor tower: enterable, with internal stairs starting from floor 1
   179→   179→function buildTower(cx: number, cz: number, w: number, d: number, floors: number, color: number, roofColor: number): MapObstacle[] {
   180→   180→  const result: MapObstacle[] = []
   181→   181→  const floorH = 3.0
   182→   182→  const wallT = 0.3
   183→   183→  const doorW = 1.5
   184→   184→  const totalH = floors * floorH
   185→   185→  // Outer walls (full height) — North, East, West
   186→   186→  result.push(ob(cx, cz - d/2, w, totalH, wallT, false, color, 'wall'))
   187→   187→  result.push(ob(cx + w/2, cz, wallT, totalH, d, false, color, 'wall'))
   188→   188→  result.push(ob(cx - w/2, cz, wallT, totalH, d, false, color, 'wall'))
   189→   189→  // South wall with door gap at ground floor only (upper floors have full wall + window)
   190→   190→  // Ground floor door
   191→   191→  result.push(ob(cx - w/4 - doorW/4, cz + d/2, w/2 - doorW/2, floorH - 0.1, wallT, false, color, 'wall', 0, false, 0))
   192→   192→  result.push(ob(cx + w/4 + doorW/4, cz + d/2, w/2 - doorW/2, floorH - 0.1, wallT, false, color, 'wall', 0, false, 0))
   193→   193→  // Upper floor walls (full, with windows)
   194→   194→  for (let f = 1; f < floors; f++) {
   195→   195→    const fy = f * floorH
   196→   196→    result.push(ob(cx, cz + d/2, w, floorH - 0.1, wallT, false, color, 'wall', 0, false, fy))
   197→   197→  }
   198→   198→  // Floor platforms for upper floors (walkable, thin slabs — no horizontal blocking)
   199→   199→  for (let f = 1; f < floors; f++) {
   200→   200→    const fy = f * floorH
   201→   201→    // platform with a gap for stairs (stair opening on one side)
   202→   202→    result.push(ob(cx + w/4, cz, w/2 - wallT, 0.2, d - wallT * 2, true, color, 'roof', 0, false, fy))
   203→   203→  }
   204→   204→  // Stairs: start from floor 1 (ground) going up to each floor
   205→   205→  // Place stairs against the north wall, going up
   206→   206→  for (let f = 0; f < floors - 1; f++) {
   207→   207→    const fy = f * floorH
   208→   208→    // 4 steps per floor (stepH=0.6, so 4 steps = 2.4 height, close to floorH=3.0)
   209→   209→    // stairs positioned at north side of tower, going from south to north (upward)
   210→   210→    result.push(...buildStairs(cx, cz - d/4, 5, 'N', 0x8a7a5a).map(s => ({ ...s, y: (s.y || 0) + fy })))
   211→   211→  }
   212→   212→  // Roof at top
   213→   213→  result.push(ob(cx, cz, w + 0.6, 0.3, d + 0.6, true, roofColor, 'roof', 0, false, totalH))
   214→   214→  return result
   215→   215→}
