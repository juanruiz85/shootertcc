    56→function buildHouse(cx: number, cz: number, w: number, d: number, h: number, color: number, roofColor: number, doorSide: 'N'|'S'|'E'|'W' = 'S'): MapObstacle[] {
    57→  const walls: MapObstacle[] = []
    58→  const wallT = 0.3
    59→  const doorW = 1.5
    60→  const doorH = 2.2  // door height
    61→  if (doorSide !== 'N') { walls.push(ob(cx, cz - d/2, w, h, wallT, false, color, 'wall')) }
    62→  else { walls.push(ob(cx - w/4 - doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx + w/4 + doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx, cz - d/2, doorW, h - doorH, wallT, false, color, 'wall', 0, true, doorH)) }
    63→  if (doorSide !== 'S') { walls.push(ob(cx, cz + d/2, w, h, wallT, false, color, 'wall')) }
    64→  else { walls.push(ob(cx - w/4 - doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx + w/4 + doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx, cz + d/2, doorW, h - doorH, wallT, false, color, 'wall', 0, true, doorH)) }
    65→  if (doorSide !== 'E') { walls.push(ob(cx + w/2, cz, wallT, h, d, false, color, 'wall')) }
    66→  else { walls.push(ob(cx + w/2, cz - d/4 - doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')); walls.push(ob(cx + w/2, cz + d/4 + doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')) }
    67→  if (doorSide !== 'W') { walls.push(ob(cx - w/2, cz, wallT, h, d, false, color, 'wall')) }
    68→  else { walls.push(ob(cx - w/2, cz - d/4 - doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')); walls.push(ob(cx - w/2, cz + d/4 + doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')) }
    69→  walls.push(ob(cx, cz, w + 0.6, 0.3, d + 0.6, true, roofColor, 'roof', 0, false, h))
    70→  return walls
    71→}

    73→function buildStairs(cx: number, cz: number, steps: number, dir: 'N'|'S'|'E'|'W', color: number): MapObstacle[] {
    74→  const result: MapObstacle[] = []
    75→  const stepH = 0.6, stepD = 0.7, stepW = 2.5
    76→  for (let i = 0; i < steps; i++) { let x = cx, z = cz; if (dir === 'N') z = cz - i * stepD; if (dir === 'S') z = cz + i * stepD; if (dir === 'E') x = cx + i * stepD; if (dir === 'W') x = cx - i * stepD; result.push(ob(x, z, stepW, stepH, stepD, true, color, 'stair', 0, false, i * stepH)) }
    77→  return result
    78→}

    80→function buildTree(cx: number, cz: number, scale: number = 1): MapObstacle[] {
    81→  const trunkH = 3 * scale
    82→  const foliageH = 2.5 * scale
    83→  return [
    84→    ob(cx, cz, 0.4 * scale, trunkH, 0.4 * scale, false, 0x7a5230, 'cyl'),
    85→    ob(cx, cz, 2.5 * scale, foliageH, 2.5 * scale, true, 0x27ae60, 'box', 0, false, trunkH),
    86→  ]
    87→}

    96→function buildTower(cx: number, cz: number, w: number, d: number, floors: number, color: number, roofColor: number): MapObstacle[] {
    97→  const result: MapObstacle[] = []
    98→  const floorH = 3.0, wallT = 0.3, doorW = 1.5, totalH = floors * floorH
    99→  result.push(ob(cx, cz - d/2, w, totalH, wallT, false, color, 'wall'))
   100→  result.push(ob(cx + w/2, cz, wallT, totalH, d, false, color, 'wall'))
   101→  result.push(ob(cx - w/2, cz, wallT, totalH, d, false, color, 'wall'))
   102→  for (let f = 0; f < floors; f++) {
   103→    const fy = f * floorH
   104→    result.push(ob(cx - w/4 - doorW/4, cz + d/2, w/2 - doorW/2, floorH - 0.1, wallT, false, color, 'wall', 0, false, fy))
   105→    result.push(ob(cx + w/4 + doorW/4, cz + d/2, w/2 - doorW/2, floorH - 0.1, wallT, false, color, 'wall', 0, false, fy))
   106→    result.push(ob(cx, cz + d/2, doorW, 0.8, wallT, false, color, 'wall', 0, true, fy + 2.0))
   107→    if (f > 0) {
   108→      result.push(ob(cx, cz, w - wallT * 2, 0.2, d - wallT * 2, true, color, 'roof', 0, false, fy))
   109→      result.push(...buildStairs(cx - w/4, cz, 4, 'N', 0x8a7a5a).map(s => ({ ...s, y: (s.y || 0) + fy })))
   110→    }
   111→  }
   112→  result.push(ob(cx, cz, w + 0.6, 0.3, d + 0.6, true, roofColor, 'roof', 0, false, totalH))
   113→  return result
   114→}

   490→function spawnBlocked(map: GameMap, x: number, z: number): boolean {
   491→  // check if a position is inside any obstacle (with margin)
   492→  // skip water and noCollide obstacles (they don't block spawning)
   493→  // skip elevated obstacles (y > 1.0): they float above ground and don't block ground spawns
   494→  const margin = 1.5
   495→  for (const o of map.obstacles) {
   496→    if (o.kind === 'water' || o.noCollide) continue
   497→    if ((o.y ?? 0) > 1.0) continue
   498→    const hw = o.w / 2, hd = o.d / 2
   499→    if (o.rotation && o.rotation !== 0) {
   500→      // rotated: use slightly larger margin (bounding-circle approximation)
   501→      const reach = Math.max(hw, hd) + margin
   502→      const dx = x - o.x, dz = z - o.z
   503→      if (dx * dx + dz * dz < reach * reach) return true
   504→    } else {
   505→      // axis-aligned box check (original behavior)
   506→      if (Math.abs(x - o.x) < hw + margin && Math.abs(z - o.z) < hd + margin) return true
   507→    }
   508→  }
   509→  return false
   510→}

   512→function pickSpawn(room: Room, preferTeam: Team = 'none'): [number, number, number] {
   513→  const map = getMap(room.mapId)
   514→  const spawns = map.spawns
   515→  // try each spawn with random offset, validate not inside obstacle
   516→  const trySpawn = (s: [number, number]): [number, number, number] => {
   517→    for (let attempt = 0; attempt < 8; attempt++) {
   518→      const x = s[0] + (Math.random() - 0.5) * 4
   519→      const z = s[1] + (Math.random() - 0.5) * 4
   520→      if (!spawnBlocked(map, x, z)) return [x, 1.7, z]
   521→    }
   522→    // fallback: use exact spawn point
   523→    return [s[0], 1.7, s[1]]
   524→  }
   525→  // PvP: bias by team (blue near spawn[0], red near spawn[1])
   526→  if (preferTeam !== 'none' && spawns.length >= 2) {
   527→    const idx = preferTeam === 'blue' ? 0 : 1
   528→    return trySpawn(spawns[idx])
   529→  }
   530→  // pick spawn farthest from any alive player
   531→  let best = spawns[0], bestD = -1
   532→  for (const s of spawns) {
   533→    let minD = Infinity
   534→    for (const p of room.players.values()) {
   535→      if (p.state !== 'alive') continue
   536→      minD = Math.min(minD, dist2D([s[0], 0, s[1]], p.pos))
   537→    }
   538→    if (minD > bestD) { bestD = minD; best = s }
   539→  }
   540→  return trySpawn(best)
   541→}

   543→function makePlayer(socketId: string, name: string, skin: string, room: Room): Player {
   544→  // team assignment: only team modes (1v1, 2v2, team) get blue/red; ffa/coop/mixed = 'none'
   545→  let team: Team = 'none'
   546→  if (room.mode === '1v1' || room.mode === '2v2' || room.mode === 'team') {
   547→    const counts = { blue: 0, red: 0 }

   549→    team = counts.blue <= counts.red ? 'blue' : 'red'

   716→function checkPvERoundEnd(r: Room) {
   717→  if (!r.roundActive) return
   718→  const aliveMobs = r.mobs.filter(m => m.state === 'alive').length
   719→  if (aliveMobs === 0) {
   720→    r.roundActive = false
   721→    r.roundEndsAt = Date.now() + ROUND_END_MS
   722→    r.nextLevel = r.level + 1
   723→    io.to(r.id).emit('room:mapChange', { mapId: r.mapId, level: r.level, mode: r.mode, banner: `¡Nivel ${r.level} superado!` })
   724→  }

   727→function checkPvPRoundEnd(r: Room) {
   728→  if (!r.roundActive || r.players.size < 2) return
   729→  const alivePlayers = Array.from(r.players.values()).filter(p => p.state === 'alive')
   730→  if (r.mode === 'ffa') {
   731→    // FFA: last man standing — round ends when ≤1 player is alive
   732→    if (alivePlayers.length <= 1) {
   733→      r.roundActive = false
   734→      r.roundEndsAt = Date.now() + ROUND_END_MS
   735→      const winnerName = alivePlayers[0]?.name ?? '—'
   736→      io.to(r.id).emit('room:mapChange', { mapId: r.mapId, level: r.level, mode: r.mode, banner: `¡${winnerName} gana la ronda!` })
   737→    }
   738→  } else {
   739→    // Team modes (1v1, 2v2, team): round ends when one team has no alive players
   740→    const teams = { blue: 0, red: 0 }
   741→    for (const p of alivePlayers) {
   742→      if (p.team === 'blue' || p.team === 'red') teams[p.team]++
   743→    }
   744→    if (teams.blue === 0 || teams.red === 0) {
   745→      r.roundActive = false
   746→      r.roundEndsAt = Date.now() + ROUND_END_MS
   747→      const winner = teams.blue > 0 ? 'Azul' : 'Rojo'
   748→      io.to(r.id).emit('room:mapChange', { mapId: r.mapId, level: r.level, mode: r.mode, banner: `¡Equipo ${winner} gana la ronda!` })
   749→    }
   750→  }
   751→}

   962→function applyDamageToPlayer(room: Room, target: Player, dmg: number, killerId: string, weapon: string, headshot: boolean, attackerPos?: [number, number, number]) {
   963→  const now = Date.now()
   964→  target.lastDamagedAt = now
   965→  // shield absorbs first
   966→  let remaining = dmg
   967→  if (target.shield > 0) {
   968→    const absorbed = Math.min(target.shield, remaining * 0.6)
   969→    target.shield -= absorbed
   970→    remaining -= absorbed
   971→  }
   972→  target.health -= remaining
   973→  io.to(room.id).emit('player:damaged', {
   974→    id: target.id, health: Math.max(0, target.health),
   975→    shield: Math.max(0, target.shield), by: killerId, headshot,
   976→    attackerPos: attackerPos ?? null,
   977→  })
   978→  if (target.health <= 0) killPlayer(room, target, killerId, weapon, headshot)
   979→}

  1000→function leaveRoom(socket: any) {
  1001→  const rid = socketToRoom.get(socket.id)
  1002→  if (!rid) return
  1003→  const room = rooms.get(rid)
  1004→  socketToRoom.delete(socket.id)
  1005→  socket.leave(rid)
  1006→  if (room) {
  1007→    const p = room.players.get(socket.id)
  1008→    room.players.delete(socket.id)
  1009→    if (p) {
  1010→      p.connected = false
  1011→      socket.to(room.id).emit('room:playerLeft', { id: socket.id })
  1012→    }
  1013→    if (room.players.size === 0) {
  1014→      room.emptySince = Date.now()

  1016→  }

  1020→function killPlayer(room: Room, victim: Player, killerId: string, weapon: string, headshot: boolean) {
  1021→  victim.state = 'dead'
  1022→  victim.health = 0
  1023→  victim.deaths += 1
  1024→  victim.streak = 0
  1025→  victim.respawnAt = Date.now() + RESPAWN_MS
  1026→  const killer = room.players.get(killerId)
  1027→  if (killer && killer.id !== victim.id) {
  1028→    killer.kills += 1
  1029→    killer.streak += 1
  1030→    killer.bestStreak = Math.max(killer.bestStreak, killer.streak)
  1031→    killer.score += headshot ? 150 : 100
  1032→    // streak rewards
  1033→    const reward = streakAt(killer.streak)
  1034→    if (reward && killer.streak === reward.streak) {
  1035→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
  1036→      io.to(killer.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
  1037→      applyStreakReward(room, killer, reward.id)
  1038→    } else {
  1039→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward: null })
  1040→    }
  1041→    // PvP modes: drop a random item on death
  1042→    if (isPvPMode(room.mode) && Math.random() < 0.6) dropItemOnMobKill(room, victim.pos)
  1043→  }
  1044→  io.to(room.id).emit('player:killed', { victimId: victim.id, killerId, weapon, headshot })
  1045→  killFeed(room, victim.name, killer?.name ?? null, weapon, headshot, 'player')
  1046→  setTimeout(() => {
  1047→    if (!victim.connected) return
  1048→    if (!room.players.has(victim.id)) return
  1049→    // Always respawn — round restarts will override with fresh state if needed
  1050→    victim.state = 'alive'
  1051→    victim.health = PLAYER_MAX_HP
  1052→    victim.shield = 0
  1053→    victim.pos = pickSpawn(room, victim.team)
  1054→    victim.weapon = 'pistol'
  1055→    victim.ammo = WEAPONS.pistol.magazine
  1056→    victim.reloading = false
  1057→    io.to(room.id).emit('player:respawned', { id: victim.id, pos: victim.pos })
  1058→  }, RESPAWN_MS)
  1059→  // round-end checks (PvP modes only — PvE round end is triggered by mob kills)
  1060→  if (isPvPMode(room.mode)) checkPvPRoundEnd(room)
  1061→}

  1063→function applyStreakReward(room: Room, p: Player, rewardId: string) {
  1064→  if (rewardId === 'heal' || rewardId === 'aura') {
  1065→    p.health = Math.min(PLAYER_MAX_HP, p.health + 25)
  1066→    p.shield = Math.min(PLAYER_MAX_SHIELD, p.shield + 25)
  1067→  }
  1068→  if (rewardId === 'bomb') {
  1069→    // damage all mobs near player + nearby enemies
  1070→    for (const m of room.mobs) {
  1071→      if (m.state !== 'alive') continue
  1072→      if (dist2D(m.pos, p.pos) < 12) {
  1073→        m.health -= 80
  1074→        m.hitFlash = Date.now()
  1075→        io.to(room.id).emit('mob:damaged', { id: m.id, health: Math.max(0, m.health), by: p.id })
  1076→        if (m.health <= 0) killMob(room, m, p.id, 'bomb', false)
  1077→      }
  1078→    }
  1079→    for (const other of room.players.values()) {
  1080→      if (other.id === p.id || other.state !== 'alive') continue
  1081→      // coop: no player damage; ffa/mixed: friendly fire on; team modes: skip same team
  1082→      if (room.mode === 'coop') continue
  1083→      if (!hasFriendlyFire(room.mode) && other.team === p.team && p.team !== 'none') continue
  1084→      if (dist2D(other.pos, p.pos) < 10) {
  1085→        applyDamageToPlayer(room, other, 60, p.id, 'bomb', false)
  1086→      }
  1087→    }
  1088→  }
  1089→  if (rewardId === 'drone') {
  1090→    room.droneOwner = p.id
  1091→    room.droneEnd = Date.now() + 20000
  1092→    io.to(room.id).emit('drone:state', { ownerId: p.id, pos: [p.pos[0], 3.5, p.pos[2]], expires: room.droneEnd, targetYaw: p.yaw })
  1093→  }
  1094→  if (rewardId === 'aura') {
  1095→    p.ammo = WEAPONS[p.weapon].magazine
  1096→    p.reloading = false
  1097→    io.to(p.id).emit('player:reloadDone', { weapon: p.weapon, ammo: p.ammo })
  1098→  }
  1099→}

  1101→function killMob(room: Room, mob: Mob, killerId: string, weapon: string, headshot: boolean) {
  1102→  mob.state = 'dead'
  1103→  mob.health = 0
  1104→  mob.respawnAt = Date.now() + MOB_RESPAWN_MS
  1105→  const killer = room.players.get(killerId)
  1106→  if (killer) {
  1107→    killer.streak += 1
  1108→    killer.bestStreak = Math.max(killer.bestStreak, killer.streak)
  1109→    killer.score += headshot ? 60 : 40
  1110→    const reward = streakAt(killer.streak)
  1111→    if (reward && killer.streak === reward.streak) {
  1112→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
  1113→      io.to(killer.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
  1114→      applyStreakReward(room, killer, reward.id)
  1115→    } else {
  1116→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward: null })
  1117→    }
  1118→  }
  1119→  io.to(room.id).emit('mob:killed', { id: mob.id, killerId, weapon, headshot })
  1120→  killFeed(room, 'Doodle Mob', killer?.name ?? null, weapon, headshot, 'mob')
  1121→  // item drop
  1122→  dropItemOnMobKill(room, mob.pos)
  1123→  checkPvERoundEnd(room)
  1124→  // schedule respawn
  1125→  setTimeout(() => {
  1126→    if (!rooms.has(room.id)) return
  1127→    if (!isPvEMode(room.mode)) return // mobs only respawn in PvE modes (coop/mixed) between levels
  1128→    mob.state = 'alive'
  1129→    mob.health = mob.maxHealth
  1130→    const a = Math.random() * Math.PI * 2
  1131→    const r = Math.random() * (HALF - 8)
  1132→    mob.pos = [Math.cos(a) * r, 1, Math.sin(a) * r]
  1133→    mob.wanderTarget = mob.pos.slice() as [number, number, number]
  1134→    io.to(room.id).emit('mob:respawned', { id: mob.id, pos: mob.pos })
  1135→  }, MOB_RESPAWN_MS)
  1136→}

  1138→// ----------------------- World tick -----------------------
  1139→setInterval(() => {
  1140→  const now = Date.now()
  1141→  // cleanup empty rooms
  1142→  for (const [rid, r] of rooms) {
  1143→    if (r.players.size === 0) {
  1144→      if (r.emptySince == null) r.emptySince = now
  1145→      if (now - (r.emptySince as number) > EMPTY_ROOM_TTL_MS) {
  1146→        // keep at least the default PvE room
  1147→        if (rooms.size > 1) {
  1148→          rooms.delete(rid)
  1149→          continue
  1150→        }
  1151→      }
  1152→    }
  1153→  }
  1154→
  1155→  for (const room of rooms.values()) {
  1156→    if (room.players.size === 0) continue
  1157→
  1158→    // round transitions
  1159→    if (!room.roundActive && room.roundEndsAt && now >= room.roundEndsAt) {
  1160→      room.roundEndsAt = 0
  1161→      if (isPvEMode(room.mode)) startPvERound(room, room.nextLevel)
  1162→      else startPvPRound(room, true)
  1163→    }
  1164→
  1165→    // mob AI (PvE modes: coop + mixed)
  1166→    if (isPvEMode(room.mode)) {
  1167→      const cfg = pveLevelConfig(room.level)
  1168→      for (const mob of room.mobs) {
  1169→        if (mob.state === 'dead') continue
  1170→        let nearest: Player | null = null
  1171→        let nd = Infinity
  1172→        for (const p of room.players.values()) {
  1173→          if (p.state !== 'alive') continue
  1174→          const d = dist2D(mob.pos, p.pos)
  1175→          if (d < nd) { nd = d; nearest = p }
  1176→        }
  1177→        let tx: number, tz: number
  1178→        if (nearest && nd < 16) {
  1179→          tx = nearest.pos[0]; tz = nearest.pos[2]
  1180→          if (nd < 1.9 && now - mob.lastAttack > MOB_ATTACK_CD) {
  1181→            mob.lastAttack = now
  1182→            applyDamageToPlayer(room, nearest, cfg.mobDamage, mob.id, 'mob', false, mob.pos)
  1183→          }
  1184→        } else {
  1185→          if (dist2D(mob.pos, mob.wanderTarget) < 1.5) {
  1186→            const a = Math.random() * Math.PI * 2
  1187→            const r = Math.random() * (HALF - 6)
  1188→            mob.wanderTarget = [Math.cos(a) * r, 1, Math.sin(a) * r]
  1189→          }
  1190→          tx = mob.wanderTarget[0]; tz = mob.wanderTarget[2]
  1191→        }
  1192→        const dx = tx - mob.pos[0], dz = tz - mob.pos[2]
  1193→        const len = Math.hypot(dx, dz) || 1
  1194→        const vx = (dx / len) * cfg.mobSpeed, vz = (dz / len) * cfg.mobSpeed
  1195→        mob.vel[0] = vx; mob.vel[1] = vz
  1196→        const step = TICK_MS / 1000
  1197→        mob.pos[0] += vx * step
  1198→        mob.pos[2] += vz * step
  1199→        clampToArena(mob.pos)
  1200→        mob.pos[1] = 1
  1201→      }
  1202→      io.to(room.id).emit('mob:state', { mobs: room.mobs.map(m => ({ id: m.id, pos: m.pos, state: m.state, flash: m.hitFlash })) })
  1203→    }
  1204→
  1205→    // PvE health regen: if player hasn't taken damage for 5s, regen 1.5 HP/sec
  1206→    if (isPvEMode(room.mode)) {
  1207→      for (const p of room.players.values()) {
  1208→        if (p.state !== 'alive') continue
  1209→        if (p.health >= PLAYER_MAX_HP) continue
  1210→        if (now - p.lastDamagedAt > 5000) {
  1211→          p.health = Math.min(PLAYER_MAX_HP, p.health + 1.5 * (TICK_MS / 1000) * 4)
  1212→          // send updated health to the player (throttled)
  1213→          if (now % 500 < TICK_MS) {
  1214→            io.to(p.id).emit('player:damaged', { id: p.id, health: Math.round(p.health), shield: p.shield, by: 'regen', headshot: false, attackerPos: null, regen: true })
  1215→          }
  1216→        }
  1217→      }
  1218→    }
  1219→
  1220→    // drone streak reward: auto-attack nearest enemy/mob
  1221→    if (room.droneOwner && now < room.droneEnd) {
  1222→      const owner = room.players.get(room.droneOwner)
  1223→      if (owner && owner.state === 'alive') {
  1224→        // attack nearest mob every ~700ms
  1225→        if (now % 700 < TICK_MS) {
  1226→          let target: Mob | null = null; let td = Infinity
  1227→          for (const m of room.mobs) {
  1228→            if (m.state !== 'alive') continue
  1229→            const d = dist2D(m.pos, owner.pos)
  1230→            if (d < 18 && d < td) { td = d; target = m }
  1231→          }
  1232→          if (target) {
  1233→            target.health -= 14
  1234→            target.hitFlash = now
  1235→            io.to(room.id).emit('mob:damaged', { id: target.id, health: Math.max(0, target.health), by: owner.id })
  1236→            if (target.health <= 0) killMob(room, target, owner.id, 'drone', false)
  1237→          }
  1238→        }
  1239→      } else if (owner && owner.state !== 'alive') {
  1240→        // drone persists but does nothing while dead
  1241→      } else {
  1242→        room.droneOwner = null
  1243→      }
  1244→    } else if (room.droneOwner && now >= room.droneEnd) {
  1245→      room.droneOwner = null
  1246→      io.to(room.id).emit('drone:state', { ownerId: null, pos: null, expires: 0 })
  1247→    }
  1248→    // emit drone position to all clients (for 3D rendering)
  1249→    if (room.droneOwner && now < room.droneEnd) {
  1250→      const owner = room.players.get(room.droneOwner)
  1251→      if (owner && owner.state === 'alive') {
  1252→        // drone hovers above and behind the owner
  1253→        const dx = -Math.sin(owner.yaw) * 3
  1254→        const dz = -Math.cos(owner.yaw) * 3
  1255→        const dronePos: [number, number, number] = [owner.pos[0] + dx, 3.5, owner.pos[2] + dz]
  1256→        io.to(room.id).emit('drone:state', { ownerId: room.droneOwner, pos: dronePos, expires: room.droneEnd, targetYaw: owner.yaw })
  1257→      }
  1258→    }
  1259→
  1260→    // item expiry
  1261→    let itemsChanged = false
  1262→    room.items = room.items.filter(it => {
  1263→      if (now - it.born > ITEM_LIFETIME_MS) { itemsChanged = true; return false }
  1264→      return true
  1265→    })
  1266→    if (itemsChanged) {
  1267→      io.to(room.id).emit('items:state', { items: room.items.map(i => ({ id: i.id, type: i.type, pos: i.pos })) })
  1268→    }
  1269→
  1270→    // periodic item spawns (PvE: every ~12s, PvP: every ~16s)
  1271→    room.itemSpawnTimer += TICK_MS
  1272→    const interval = isPvEMode(room.mode) ? ITEM_RESPAWN_INTERVAL_MS : 16000
  1273→    if (room.itemSpawnTimer >= interval && room.items.length < 6) {
  1274→      room.itemSpawnTimer = 0
  1275→      const roll = Math.random()
  1276→      const type: 'ammo'|'heal'|'shield' = roll < 0.5 ? 'ammo' : roll < 0.8 ? 'heal' : 'shield'
  1277→      spawnItem(room, type)
  1278→    }
  1279→
  1280→    // roster every ~1s
  1281→    if (now % 1000 < TICK_MS) emitRoomState(room)
  1282→  }
  1283→}, TICK_MS)
