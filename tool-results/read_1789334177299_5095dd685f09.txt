    61→export function streakAt(streak: number): KillstreakDef | null {
    62→  let best: KillstreakDef | null = null
    63→  for (const k of KILLSTREAKS) if (streak >= k.streak && (!best || k.streak > best.streak)) best = k
    64→  return best
    65→}

   103→// helper to build obstacle
   104→const ob = (x: number, z: number, w: number, h: number, d: number, climbable = true, color = C_WOOD, kind: 'box'|'cyl'|'ramp'|'wall'|'stair'|'water'|'roof' = 'box', rotation = 0, noCollide = false, y = 0): MapObstacle => ({ x, z, w, h, d, climbable, color, kind, rotation, noCollide, y })

   106→// Build a house: 4 walls with a door gap on one side, windows, optional roof
   107→// Returns array of obstacles (wall segments)
   108→function buildHouse(cx: number, cz: number, w: number, d: number, h: number, color: number, roofColor: number, doorSide: 'N'|'S'|'E'|'W' = 'S'): MapObstacle[] {
   109→  const walls: MapObstacle[] = []
   110→  const wallT = 0.3
   111→  const doorW = 1.5
   112→  const doorH = 2.2  // door height
   113→  const winH = 1.0
   114→  // North wall (z = cz - d/2)
   115→  if (doorSide !== 'N') {
   116→    walls.push(ob(cx, cz - d/2, w, h, wallT, false, color, 'wall'))
   117→  } else {
   118→    walls.push(ob(cx - w/4 - doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall'))
   119→    walls.push(ob(cx + w/4 + doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall'))
   120→    // window above door (positioned at doorH, no collision)
   121→    walls.push(ob(cx, cz - d/2, doorW, h - doorH, wallT, false, color, 'wall', 0, true, doorH))
   122→  }
   123→  // South wall (z = cz + d/2)
   124→  if (doorSide !== 'S') {
   125→    walls.push(ob(cx, cz + d/2, w, h, wallT, false, color, 'wall'))
   126→  } else {
   127→    walls.push(ob(cx - w/4 - doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall'))
   128→    walls.push(ob(cx + w/4 + doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall'))
   129→    walls.push(ob(cx, cz + d/2, doorW, h - doorH, wallT, false, color, 'wall', 0, true, doorH))
   130→  }
   131→  // East wall (x = cx + w/2)
   132→  if (doorSide !== 'E') {
   133→    walls.push(ob(cx + w/2, cz, wallT, h, d, false, color, 'wall'))
   134→  } else {
   135→    walls.push(ob(cx + w/2, cz - d/4 - doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall'))
   136→    walls.push(ob(cx + w/2, cz + d/4 + doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall'))
   137→  }
   138→  // West wall (x = cx - w/2)
   139→  if (doorSide !== 'W') {
   140→    walls.push(ob(cx - w/2, cz, wallT, h, d, false, color, 'wall'))
   141→  } else {
   142→    walls.push(ob(cx - w/2, cz - d/4 - doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall'))
   143→    walls.push(ob(cx - w/2, cz + d/4 + doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall'))
   144→  }
   145→  // Roof at top of walls (player can stand on it)
   146→  walls.push(ob(cx, cz, w + 0.6, 0.3, d + 0.6, true, roofColor, 'roof', 0, false, h))
   147→  return walls
   148→}

   150→// Build stairs: a series of steps going up, each at increasing height
   151→function buildStairs(cx: number, cz: number, steps: number, dir: 'N'|'S'|'E'|'W', color: number): MapObstacle[] {
   152→  const result: MapObstacle[] = []
   153→  const stepH = 0.6, stepD = 0.7, stepW = 2.5
   154→  for (let i = 0; i < steps; i++) {
   155→    let x = cx, z = cz
   156→    if (dir === 'N') z = cz - i * stepD
   157→    if (dir === 'S') z = cz + i * stepD
   158→    if (dir === 'E') x = cx + i * stepD
   159→    if (dir === 'W') x = cx - i * stepD
   160→    // each step is positioned at y = i * stepH (stacking upward)
   161→    result.push(ob(x, z, stepW, stepH, stepD, true, color, 'stair', 0, false, i * stepH))
   162→  }
   163→  return result
   164→}
   165→
   166→// Build a tree (trunk + foliage on top)
   167→function buildTree(cx: number, cz: number, scale: number = 1): MapObstacle[] {
   168→  const trunkH = 3 * scale
   169→  const foliageH = 2.5 * scale
   170→  return [
   171→    // trunk at ground level
   172→    ob(cx, cz, 0.4 * scale, trunkH, 0.4 * scale, false, 0x7a5230, 'cyl'),
   173→    // foliage on top of trunk
   174→    ob(cx, cz, 2.5 * scale, foliageH, 2.5 * scale, true, 0x27ae60, 'box', 0, false, trunkH),
   175→  ]
   176→}
   177→
   178→// Build a car (body + cabin on top)
   179→function buildCar(cx: number, cz: number, color: number, rotation: number = 0): MapObstacle[] {
   180→  return [
   181→    ob(cx, cz, 2, 0.8, 4, true, color, 'box', rotation),           // body at ground
   182→    ob(cx, cz, 1.8, 0.6, 2, false, 0x2c3e50, 'box', rotation, false, 0.8), // cabin on top
   183→  ]
   184→}

   186→// Build a multi-floor tower: enterable, with internal stairs starting from floor 1
   187→function buildTower(cx: number, cz: number, w: number, d: number, floors: number, color: number, roofColor: number): MapObstacle[] {
   188→  const result: MapObstacle[] = []
   189→  const floorH = 3.0
   190→  const wallT = 0.3
   191→  const doorW = 1.5
   192→  const totalH = floors * floorH
   193→  // Outer walls (full height) — North, East, West
   194→  result.push(ob(cx, cz - d/2, w, totalH, wallT, false, color, 'wall'))
   195→  result.push(ob(cx + w/2, cz, wallT, totalH, d, false, color, 'wall'))
   196→  result.push(ob(cx - w/2, cz, wallT, totalH, d, false, color, 'wall'))
   197→  // South wall with door gap at ground floor only (upper floors have full wall + window)
   198→  // Ground floor door
   199→  result.push(ob(cx - w/4 - doorW/4, cz + d/2, w/2 - doorW/2, floorH - 0.1, wallT, false, color, 'wall', 0, false, 0))
   200→  result.push(ob(cx + w/4 + doorW/4, cz + d/2, w/2 - doorW/2, floorH - 0.1, wallT, false, color, 'wall', 0, false, 0))
   201→  // Upper floor walls (full, with windows)
   202→  for (let f = 1; f < floors; f++) {
   203→    const fy = f * floorH
   204→    result.push(ob(cx, cz + d/2, w, floorH - 0.1, wallT, false, color, 'wall', 0, false, fy))
   205→  }
   206→  // Floor platforms for upper floors (walkable, thin slabs — no horizontal blocking)
   207→  for (let f = 1; f < floors; f++) {
   208→    const fy = f * floorH
   209→    // platform with a gap for stairs (stair opening on one side)
   210→    result.push(ob(cx + w/4, cz, w/2 - wallT, 0.2, d - wallT * 2, true, color, 'roof', 0, false, fy))
   211→  }
   212→  // Stairs: start from floor 1 (ground) going up to each floor
   213→  // Place stairs against the north wall, going up
   214→  for (let f = 0; f < floors - 1; f++) {
   215→    const fy = f * floorH
   216→    // 4 steps per floor (stepH=0.6, so 4 steps = 2.4 height, close to floorH=3.0)
   217→    // stairs positioned at north side of tower, going from south to north (upward)
   218→    result.push(...buildStairs(cx, cz - d/4, 5, 'N', 0x8a7a5a).map(s => ({ ...s, y: (s.y || 0) + fy })))
   219→  }
   220→  // Roof at top
   221→  result.push(ob(cx, cz, w + 0.6, 0.3, d + 0.6, true, roofColor, 'roof', 0, false, totalH))
   222→  return result
   223→}
