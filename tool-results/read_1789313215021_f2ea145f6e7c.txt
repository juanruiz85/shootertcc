    56→function buildHouse(cx: number, cz: number, w: number, d: number, h: number, color: number, roofColor: number, doorSide: 'N'|'S'|'E'|'W' = 'S'): MapObstacle[] {
    57→  const walls: MapObstacle[] = []
    58→  const wallT = 0.3
    59→  const doorW = 1.5
    60→  const doorH = 2.2  // door height
    61→  if (doorSide !== 'N') { walls.push(ob(cx, cz - d/2, w, h, wallT, false, color, 'wall')) }
    62→  else { walls.push(ob(cx - w/4 - doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx + w/4 + doorW/4, cz - d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx, cz - d/2, doorW, h - doorH, wallT, false, color, 'wall', 0, true, doorH)) }
    63→  if (doorSide !== 'S') { walls.push(ob(cx, cz + d/2, w, h, wallT, false, color, 'wall')) }
    64→  else { walls.push(ob(cx - w/4 - doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx + w/4 + doorW/4, cz + d/2, w/2 - doorW/2, h, wallT, false, color, 'wall')); walls.push(ob(cx, cz + d/2, doorW, h - doorH, wallT, false, color, 'wall', 0, true, doorH)) }
    65→  if (doorSide !== 'E') { walls.push(ob(cx + w/2, cz, wallT, h, d, false, color, 'wall')) }
    66→  else { walls.push(ob(cx + w/2, cz - d/4 - doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')); walls.push(ob(cx + w/2, cz + d/4 + doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')) }
    67→  if (doorSide !== 'W') { walls.push(ob(cx - w/2, cz, wallT, h, d, false, color, 'wall')) }
    68→  else { walls.push(ob(cx - w/2, cz - d/4 - doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')); walls.push(ob(cx - w/2, cz + d/4 + doorW/4, wallT, h, d/2 - doorW/2, false, color, 'wall')) }
    69→  walls.push(ob(cx, cz, w + 0.6, 0.3, d + 0.6, true, roofColor, 'roof', 0, false, h))
    70→  return walls
    71→}

    73→function buildStairs(cx: number, cz: number, steps: number, dir: 'N'|'S'|'E'|'W', color: number): MapObstacle[] {
    74→  const result: MapObstacle[] = []
    75→  const stepH = 0.6, stepD = 0.7, stepW = 2.5
    76→  for (let i = 0; i < steps; i++) { let x = cx, z = cz; if (dir === 'N') z = cz - i * stepD; if (dir === 'S') z = cz + i * stepD; if (dir === 'E') x = cx + i * stepD; if (dir === 'W') x = cx - i * stepD; result.push(ob(x, z, stepW, stepH, stepD, true, color, 'stair', 0, false, i * stepH)) }
    77→  return result
    78→}

    80→function buildTree(cx: number, cz: number, scale: number = 1): MapObstacle[] {
    81→  const trunkH = 3 * scale
    82→  const foliageH = 2.5 * scale
    83→  return [
    84→    ob(cx, cz, 0.4 * scale, trunkH, 0.4 * scale, false, 0x7a5230, 'cyl'),
    85→    ob(cx, cz, 2.5 * scale, foliageH, 2.5 * scale, true, 0x27ae60, 'box', 0, false, trunkH),
    86→  ]
    87→}

    96→function buildTower(cx: number, cz: number, w: number, d: number, floors: number, color: number, roofColor: number): MapObstacle[] {
    97→  const result: MapObstacle[] = []
    98→  const floorH = 3.0
    99→  const wallT = 0.3
   100→  const doorW = 1.5
   101→  const totalH = floors * floorH
   102→  // Outer walls (full height) — North, East, West
   103→  result.push(ob(cx, cz - d/2, w, totalH, wallT, false, color, 'wall'))
   104→  result.push(ob(cx + w/2, cz, wallT, totalH, d, false, color, 'wall'))
   105→  result.push(ob(cx - w/2, cz, wallT, totalH, d, false, color, 'wall'))
   106→  // South wall with door gap at ground floor only
   107→  result.push(ob(cx - w/4 - doorW/4, cz + d/2, w/2 - doorW/2, floorH - 0.1, wallT, false, color, 'wall', 0, false, 0))
   108→  result.push(ob(cx + w/4 + doorW/4, cz + d/2, w/2 - doorW/2, floorH - 0.1, wallT, false, color, 'wall', 0, false, 0))
   109→  // Upper floor walls (full)
   110→  for (let f = 1; f < floors; f++) {
   111→    const fy = f * floorH
   112→    result.push(ob(cx, cz + d/2, w, floorH - 0.1, wallT, false, color, 'wall', 0, false, fy))
   113→  }
   114→  // Floor platforms for upper floors (walkable, thin slabs)
   115→  for (let f = 1; f < floors; f++) {
   116→    const fy = f * floorH
   117→    result.push(ob(cx + w/4, cz, w/2 - wallT, 0.2, d - wallT * 2, true, color, 'roof', 0, false, fy))
   118→  }
   119→  // Stairs: start from floor 1 (ground) going up to each floor
   120→  for (let f = 0; f < floors - 1; f++) {
   121→    const fy = f * floorH
   122→    result.push(...buildStairs(cx, cz - d/4, 5, 'N', 0x8a7a5a).map(s => ({ ...s, y: (s.y || 0) + fy })))
   123→  }
   124→  // Roof at top
   125→  result.push(ob(cx, cz, w + 0.6, 0.3, d + 0.6, true, roofColor, 'roof', 0, false, totalH))
   126→  return result
   127→}

   503→function spawnBlocked(map: GameMap, x: number, z: number): boolean {
   504→  // check if a position is inside any obstacle (with margin)
   505→  // skip water and noCollide obstacles (they don't block spawning)
   506→  // skip elevated obstacles (y > 1.0): they float above ground and don't block ground spawns
   507→  const margin = 1.5
   508→  for (const o of map.obstacles) {
   509→    if (o.kind === 'water' || o.noCollide) continue
   510→    if ((o.y ?? 0) > 1.0) continue
   511→    const hw = o.w / 2, hd = o.d / 2
   512→    if (o.rotation && o.rotation !== 0) {
   513→      // rotated: use slightly larger margin (bounding-circle approximation)
   514→      const reach = Math.max(hw, hd) + margin
   515→      const dx = x - o.x, dz = z - o.z
   516→      if (dx * dx + dz * dz < reach * reach) return true
   517→    } else {
   518→      // axis-aligned box check (original behavior)
   519→      if (Math.abs(x - o.x) < hw + margin && Math.abs(z - o.z) < hd + margin) return true
   520→    }
   521→  }
   522→  return false
   523→}

   525→function pickSpawn(room: Room, preferTeam: Team = 'none'): [number, number, number] {
   526→  const map = getMap(room.mapId)
   527→  const spawns = map.spawns
   528→  // try each spawn with random offset, validate not inside obstacle
   529→  const trySpawn = (s: [number, number]): [number, number, number] => {
   530→    for (let attempt = 0; attempt < 8; attempt++) {
   531→      const x = s[0] + (Math.random() - 0.5) * 4
   532→      const z = s[1] + (Math.random() - 0.5) * 4
   533→      if (!spawnBlocked(map, x, z)) return [x, 1.7, z]
   534→    }
   535→    // fallback: use exact spawn point
   536→    return [s[0], 1.7, s[1]]
   537→  }
   538→  // PvP: bias by team (blue near spawn[0], red near spawn[1])
   539→  if (preferTeam !== 'none' && spawns.length >= 2) {
   540→    const idx = preferTeam === 'blue' ? 0 : 1
   541→    return trySpawn(spawns[idx])
   542→  }
   543→  // pick spawn farthest from any alive player
   544→  let best = spawns[0], bestD = -1
   545→  for (const s of spawns) {
   546→    let minD = Infinity
   547→    for (const p of room.players.values()) {
   548→      if (p.state !== 'alive') continue
   549→      minD = Math.min(minD, dist2D([s[0], 0, s[1]], p.pos))
   550→    }
   551→    if (minD > bestD) { bestD = minD; best = s }
   552→  }
   553→  return trySpawn(best)
   554→}

   556→function makePlayer(socketId: string, name: string, skin: string, room: Room): Player {
   557→  // team assignment: only team modes (1v1, 2v2, team) get blue/red; ffa/coop/mixed = 'none'
   558→  let team: Team = 'none'
   559→  if (room.mode === '1v1' || room.mode === '2v2' || room.mode === 'team') {
   560→    const counts = { blue: 0, red: 0 }
   561→    for (const p of room.players.values()) counts[p.team]++
   562→    team = counts.blue <= counts.red ? 'blue' : 'red'

   740→function checkPvPRoundEnd(r: Room) {
   741→  if (!r.roundActive || r.players.size < 2) return
   742→  const alivePlayers = Array.from(r.players.values()).filter(p => p.state === 'alive')
   743→  if (r.mode === 'ffa') {
   744→    // FFA: last man standing — round ends when ≤1 player is alive
   745→    if (alivePlayers.length <= 1) {
   746→      r.roundActive = false
   747→      r.roundEndsAt = Date.now() + ROUND_END_MS
   748→      const winnerName = alivePlayers[0]?.name ?? '—'
   749→      io.to(r.id).emit('room:mapChange', { mapId: r.mapId, level: r.level, mode: r.mode, banner: `¡${winnerName} gana la ronda!` })
   750→    }
   751→  } else {
   752→    // Team modes (1v1, 2v2, team): round ends when one team has no alive players
   753→    const teams = { blue: 0, red: 0 }
   754→    for (const p of alivePlayers) {
   755→      if (p.team === 'blue' || p.team === 'red') teams[p.team]++
   756→    }
   757→    if (teams.blue === 0 || teams.red === 0) {
   758→      r.roundActive = false
   759→      r.roundEndsAt = Date.now() + ROUND_END_MS
   760→      const winner = teams.blue > 0 ? 'Azul' : 'Rojo'
   761→      io.to(r.id).emit('room:mapChange', { mapId: r.mapId, level: r.level, mode: r.mode, banner: `¡Equipo ${winner} gana la ronda!` })
   762→    }
   763→  }
   764→}

   975→function applyDamageToPlayer(room: Room, target: Player, dmg: number, killerId: string, weapon: string, headshot: boolean, attackerPos?: [number, number, number]) {
   976→  const now = Date.now()
   977→  target.lastDamagedAt = now
   978→  // shield absorbs first
   979→  let remaining = dmg
   980→  if (target.shield > 0) {
   981→    const absorbed = Math.min(target.shield, remaining * 0.6)
   982→    target.shield -= absorbed
   983→    remaining -= absorbed
   984→  }
   985→  target.health -= remaining
   986→  io.to(room.id).emit('player:damaged', {
   987→    id: target.id, health: Math.max(0, target.health),
   988→    shield: Math.max(0, target.shield), by: killerId, headshot,
   989→    attackerPos: attackerPos ?? null,
   990→  })
   991→  if (target.health <= 0) killPlayer(room, target, killerId, weapon, headshot)

   994→function joinRoom(socket: any, room: Room, name: string, skin: string) {
   995→  leaveRoom(socket)
   996→  const player = makePlayer(socket.id, name, skin, room)

  1004→    mobs: room.mobs.map(m => ({ id: m.id, pos: m.pos, state: m.state, health: m.health })),

  1008→  room.emptySince = null

  1013→function leaveRoom(socket: any) {
  1014→  const rid = socketToRoom.get(socket.id)
  1015→  if (!rid) return
  1016→  const room = rooms.get(rid)
  1017→  socketToRoom.delete(socket.id)
  1018→  socket.leave(rid)
  1019→  if (room) {
  1020→    const p = room.players.get(socket.id)
  1021→    room.players.delete(socket.id)
  1022→    if (p) {
  1023→      p.connected = false
  1024→      socket.to(room.id).emit('room:playerLeft', { id: socket.id })
  1025→    }
  1026→    if (room.players.size === 0) {
  1027→      room.emptySince = Date.now()
  1028→    }
  1029→  }
  1030→  broadcastLobbyState()

  1033→function killPlayer(room: Room, victim: Player, killerId: string, weapon: string, headshot: boolean) {
  1034→  victim.state = 'dead'
  1035→  victim.health = 0
  1036→  victim.deaths += 1
  1037→  victim.streak = 0
  1038→  victim.respawnAt = Date.now() + RESPAWN_MS
  1039→  const killer = room.players.get(killerId)
  1040→  if (killer && killer.id !== victim.id) {
  1041→    killer.kills += 1
  1042→    killer.streak += 1
  1043→    killer.bestStreak = Math.max(killer.bestStreak, killer.streak)
  1044→    killer.score += headshot ? 150 : 100
  1045→    // streak rewards
  1046→    const reward = streakAt(killer.streak)
  1047→    if (reward && killer.streak === reward.streak) {
  1048→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
  1049→      io.to(killer.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
  1050→      applyStreakReward(room, killer, reward.id)
  1051→    } else {
  1052→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward: null })
  1053→    }
  1054→    // PvP modes: drop a random item on death
  1055→    if (isPvPMode(room.mode) && Math.random() < 0.6) dropItemOnMobKill(room, victim.pos)
  1056→  }
  1057→  io.to(room.id).emit('player:killed', { victimId: victim.id, killerId, weapon, headshot })
  1058→  killFeed(room, victim.name, killer?.name ?? null, weapon, headshot, 'player')
  1059→  setTimeout(() => {
  1060→    if (!victim.connected) return
  1061→    if (!room.players.has(victim.id)) return
  1062→    // Always respawn — round restarts will override with fresh state if needed
  1063→    victim.state = 'alive'
  1064→    victim.health = PLAYER_MAX_HP
  1065→    victim.shield = 0
  1066→    victim.pos = pickSpawn(room, victim.team)
  1067→    victim.weapon = 'pistol'
  1068→    victim.ammo = WEAPONS.pistol.magazine
  1069→    victim.reloading = false
  1070→    io.to(room.id).emit('player:respawned', { id: victim.id, pos: victim.pos })
  1071→  }, RESPAWN_MS)
  1072→  // round-end checks (PvP modes only — PvE round end is triggered by mob kills)
  1073→  if (isPvPMode(room.mode)) checkPvPRoundEnd(room)
  1074→}

  1076→function applyStreakReward(room: Room, p: Player, rewardId: string) {
  1077→  if (rewardId === 'heal' || rewardId === 'aura') {
  1078→    p.health = Math.min(PLAYER_MAX_HP, p.health + 25)
  1079→    p.shield = Math.min(PLAYER_MAX_SHIELD, p.shield + 25)
  1080→  }
  1081→  if (rewardId === 'bomb') {
  1082→    // damage all mobs near player + nearby enemies
  1083→    for (const m of room.mobs) {
  1084→      if (m.state !== 'alive') continue
  1085→      if (dist2D(m.pos, p.pos) < 12) {
  1086→        m.health -= 80
  1087→        m.hitFlash = Date.now()
  1088→        io.to(room.id).emit('mob:damaged', { id: m.id, health: Math.max(0, m.health), by: p.id })
  1089→        if (m.health <= 0) killMob(room, m, p.id, 'bomb', false)
  1090→      }
  1091→    }
  1092→    for (const other of room.players.values()) {
  1093→      if (other.id === p.id || other.state !== 'alive') continue
  1094→      // coop: no player damage; ffa/mixed: friendly fire on; team modes: skip same team
  1095→      if (room.mode === 'coop') continue
  1096→      if (!hasFriendlyFire(room.mode) && other.team === p.team && p.team !== 'none') continue
  1097→      if (dist2D(other.pos, p.pos) < 10) {
  1098→        applyDamageToPlayer(room, other, 60, p.id, 'bomb', false)
  1099→      }
  1100→    }
  1101→  }
  1102→  if (rewardId === 'drone') {
  1103→    room.droneOwner = p.id
  1104→    room.droneEnd = Date.now() + 20000
  1105→    io.to(room.id).emit('drone:state', { ownerId: p.id, pos: [p.pos[0], 3.5, p.pos[2]], expires: room.droneEnd, targetYaw: p.yaw })
  1106→  }
  1107→  if (rewardId === 'aura') {
  1108→    p.ammo = WEAPONS[p.weapon].magazine
  1109→    p.reloading = false
  1110→    io.to(p.id).emit('player:reloadDone', { weapon: p.weapon, ammo: p.ammo })
  1111→  }
  1112→}

  1114→function killMob(room: Room, mob: Mob, killerId: string, weapon: string, headshot: boolean) {
  1115→  mob.state = 'dead'
  1116→  mob.health = 0
  1117→  mob.respawnAt = Date.now() + MOB_RESPAWN_MS
  1118→  const killer = room.players.get(killerId)
  1119→  if (killer) {
  1120→    killer.streak += 1
  1121→    killer.bestStreak = Math.max(killer.bestStreak, killer.streak)
  1122→    killer.score += headshot ? 60 : 40
  1123→    const reward = streakAt(killer.streak)
  1124→    if (reward && killer.streak === reward.streak) {
  1125→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
  1126→      io.to(killer.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward })
  1127→      applyStreakReward(room, killer, reward.id)
  1128→    } else {
  1129→      io.to(room.id).emit('player:streak', { id: killer.id, streak: killer.streak, reward: null })
  1130→    }
  1131→  }
  1132→  io.to(room.id).emit('mob:killed', { id: mob.id, killerId, weapon, headshot })
  1133→  killFeed(room, 'Doodle Mob', killer?.name ?? null, weapon, headshot, 'mob')
  1134→  // item drop
  1135→  dropItemOnMobKill(room, mob.pos)
  1136→  checkPvERoundEnd(room)
  1137→  // schedule respawn
  1138→  setTimeout(() => {
  1139→    if (!rooms.has(room.id)) return
  1140→    if (!isPvEMode(room.mode)) return // mobs only respawn in PvE modes (coop/mixed) between levels
  1141→    mob.state = 'alive'
  1142→    mob.health = mob.maxHealth
  1143→    const a = Math.random() * Math.PI * 2
  1144→    const r = Math.random() * (HALF - 8)
  1145→    mob.pos = [Math.cos(a) * r, 1, Math.sin(a) * r]
  1146→    mob.wanderTarget = mob.pos.slice() as [number, number, number]
  1147→    io.to(room.id).emit('mob:respawned', { id: mob.id, pos: mob.pos })
  1148→  }, MOB_RESPAWN_MS)
  1149→}
  1150→
  1151→// ----------------------- World tick -----------------------
  1152→setInterval(() => {
  1153→  const now = Date.now()
  1154→  // cleanup empty rooms
  1155→  for (const [rid, r] of rooms) {
  1156→    if (r.players.size === 0) {
  1157→      if (r.emptySince == null) r.emptySince = now
  1158→      if (now - (r.emptySince as number) > EMPTY_ROOM_TTL_MS) {
  1159→        // keep at least the default PvE room
  1160→        if (rooms.size > 1) {
  1161→          rooms.delete(rid)
  1162→          continue
  1163→        }
  1164→      }
  1165→    }
  1166→  }
  1167→
  1168→  for (const room of rooms.values()) {
  1169→    if (room.players.size === 0) continue
  1170→
  1171→    // round transitions
  1172→    if (!room.roundActive && room.roundEndsAt && now >= room.roundEndsAt) {
  1173→      room.roundEndsAt = 0
  1174→      if (isPvEMode(room.mode)) startPvERound(room, room.nextLevel)
  1175→      else startPvPRound(room, true)
  1176→    }
  1177→
  1178→    // mob AI (PvE modes: coop + mixed)
  1179→    if (isPvEMode(room.mode)) {
  1180→      const cfg = pveLevelConfig(room.level)
  1181→      for (const mob of room.mobs) {
  1182→        if (mob.state === 'dead') continue
  1183→        let nearest: Player | null = null
  1184→        let nd = Infinity
  1185→        for (const p of room.players.values()) {
  1186→          if (p.state !== 'alive') continue
  1187→          const d = dist2D(mob.pos, p.pos)
  1188→          if (d < nd) { nd = d; nearest = p }
  1189→        }
  1190→        let tx: number, tz: number
  1191→        if (nearest && nd < 16) {
  1192→          tx = nearest.pos[0]; tz = nearest.pos[2]
  1193→          if (nd < 1.9 && now - mob.lastAttack > MOB_ATTACK_CD) {
  1194→            mob.lastAttack = now
  1195→            applyDamageToPlayer(room, nearest, cfg.mobDamage, mob.id, 'mob', false, mob.pos)
  1196→          }
  1197→        } else {
  1198→          if (dist2D(mob.pos, mob.wanderTarget) < 1.5) {
  1199→            const a = Math.random() * Math.PI * 2
  1200→            const r = Math.random() * (HALF - 6)
  1201→            mob.wanderTarget = [Math.cos(a) * r, 1, Math.sin(a) * r]
  1202→          }
  1203→          tx = mob.wanderTarget[0]; tz = mob.wanderTarget[2]
  1204→        }
  1205→        const dx = tx - mob.pos[0], dz = tz - mob.pos[2]
  1206→        const len = Math.hypot(dx, dz) || 1
  1207→        const vx = (dx / len) * cfg.mobSpeed, vz = (dz / len) * cfg.mobSpeed
  1208→        mob.vel[0] = vx; mob.vel[1] = vz
  1209→        const step = TICK_MS / 1000
  1210→        mob.pos[0] += vx * step
  1211→        mob.pos[2] += vz * step
  1212→        clampToArena(mob.pos)
  1213→        mob.pos[1] = 1
  1214→      }
  1215→      io.to(room.id).emit('mob:state', { mobs: room.mobs.map(m => ({ id: m.id, pos: m.pos, state: m.state, flash: m.hitFlash })) })
  1216→    }
  1217→
  1218→    // PvE health regen: if player hasn't taken damage for 5s, regen 1.5 HP/sec
  1219→    if (isPvEMode(room.mode)) {
  1220→      for (const p of room.players.values()) {
  1221→        if (p.state !== 'alive') continue
  1222→        if (p.health >= PLAYER_MAX_HP) continue
  1223→        if (now - p.lastDamagedAt > 5000) {
  1224→          p.health = Math.min(PLAYER_MAX_HP, p.health + 1.5 * (TICK_MS / 1000) * 4)
  1225→          // send updated health to the player (throttled)
  1226→          if (now % 500 < TICK_MS) {
  1227→            io.to(p.id).emit('player:damaged', { id: p.id, health: Math.round(p.health), shield: p.shield, by: 'regen', headshot: false, attackerPos: null, regen: true })
  1228→          }
  1229→        }
  1230→      }
  1231→    }
  1232→
  1233→    // drone streak reward: auto-attack nearest enemy/mob
  1234→    if (room.droneOwner && now < room.droneEnd) {
  1235→      const owner = room.players.get(room.droneOwner)
  1236→      if (owner && owner.state === 'alive') {
  1237→        // attack nearest mob every ~700ms
  1238→        if (now % 700 < TICK_MS) {
  1239→          let target: Mob | null = null; let td = Infinity
  1240→          for (const m of room.mobs) {
  1241→            if (m.state !== 'alive') continue
  1242→            const d = dist2D(m.pos, owner.pos)
  1243→            if (d < 18 && d < td) { td = d; target = m }
  1244→          }
  1245→          if (target) {
  1246→            target.health -= 14
  1247→            target.hitFlash = now
  1248→            io.to(room.id).emit('mob:damaged', { id: target.id, health: Math.max(0, target.health), by: owner.id })
  1249→            if (target.health <= 0) killMob(room, target, owner.id, 'drone', false)
  1250→          }
  1251→        }
  1252→      } else if (owner && owner.state !== 'alive') {
  1253→        // drone persists but does nothing while dead
  1254→      } else {
  1255→        room.droneOwner = null
  1256→      }
  1257→    } else if (room.droneOwner && now >= room.droneEnd) {
  1258→      room.droneOwner = null
  1259→      io.to(room.id).emit('drone:state', { ownerId: null, pos: null, expires: 0 })
  1260→    }
  1261→    // emit drone position to all clients (for 3D rendering)
  1262→    if (room.droneOwner && now < room.droneEnd) {
  1263→      const owner = room.players.get(room.droneOwner)
  1264→      if (owner && owner.state === 'alive') {
  1265→        // drone hovers above and behind the owner
  1266→        const dx = -Math.sin(owner.yaw) * 3
  1267→        const dz = -Math.cos(owner.yaw) * 3
  1268→        const dronePos: [number, number, number] = [owner.pos[0] + dx, 3.5, owner.pos[2] + dz]
  1269→        io.to(room.id).emit('drone:state', { ownerId: room.droneOwner, pos: dronePos, expires: room.droneEnd, targetYaw: owner.yaw })
  1270→      }
  1271→    }
  1272→
  1273→    // item expiry
  1274→    let itemsChanged = false
  1275→    room.items = room.items.filter(it => {
  1276→      if (now - it.born > ITEM_LIFETIME_MS) { itemsChanged = true; return false }
  1277→      return true
  1278→    })
  1279→    if (itemsChanged) {
  1280→      io.to(room.id).emit('items:state', { items: room.items.map(i => ({ id: i.id, type: i.type, pos: i.pos })) })
  1281→    }
  1282→
  1283→    // periodic item spawns (PvE: every ~12s, PvP: every ~16s)
  1284→    room.itemSpawnTimer += TICK_MS
  1285→    const interval = isPvEMode(room.mode) ? ITEM_RESPAWN_INTERVAL_MS : 16000
  1286→    if (room.itemSpawnTimer >= interval && room.items.length < 6) {
  1287→      room.itemSpawnTimer = 0
  1288→      const roll = Math.random()
  1289→      const type: 'ammo'|'heal'|'shield' = roll < 0.5 ? 'ammo' : roll < 0.8 ? 'heal' : 'shield'
  1290→      spawnItem(room, type)
  1291→    }
  1292→
  1293→    // roster every ~1s
  1294→    if (now % 1000 < TICK_MS) emitRoomState(room)
  1295→  }
  1296→}, TICK_MS)
