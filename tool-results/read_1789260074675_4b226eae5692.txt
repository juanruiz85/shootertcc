   319→        function createDoodleMesh(geometry, color, scale = 1) {
   320→            const group = new THREE.Group();
   321→            const material = new THREE.MeshToonMaterial({ color: color });
   322→            const mesh = new THREE.Mesh(geometry, material);
   323→            const edges = new THREE.EdgesGeometry(geometry, 15);
   324→            const lineMaterial = new THREE.LineBasicMaterial({ color: 0x000000, linewidth: 2 });
   325→            const wireframe = new THREE.LineSegments(edges, lineMaterial);
   326→            wireframe.scale.set(1.01, 1.01, 1.01);
   327→            group.add(mesh);
   328→            group.add(wireframe);
   329→            group.scale.set(scale, scale, scale);
   330→            group.userData = { 
   331→                wireframe: wireframe,
   332→                wobbleSpeed: Math.random() * 0.003 + 0.002,
   333→                wobbleOffset: Math.random() * 100,
   334→                isDoodle: true
   335→            };
   336→            return group;
   337→        }

   339→        class AABB {
   340→            constructor(min, max) { this.min = min; this.max = max; }
   341→            intersects(other) {
   342→                return this.min.x <= other.max.x && this.max.x >= other.min.x &&
   343→                       this.min.y <= other.max.y && this.max.y >= other.min.y &&
   344→                       this.min.z <= other.max.z && this.max.z >= other.min.z;
   345→            }
   346→            contains(point) {
   347→                return point.x >= this.min.x && point.x <= this.max.x &&
   348→                       point.y >= this.min.y && point.y <= this.max.y &&
   349→                       point.z >= this.min.z && point.z <= this.max.z;
   350→            }
   351→        }

   353→        class Entity {
   354→            constructor(position, type) {
   355→                this.position = position.clone();
   356→                this.velocity = new THREE.Vector3();
   357→                this.type = type;
   358→                this.alive = true;
   359→                this.mesh = null;
   360→                this.bbox = null;
   361→            }
   362→            update(delta) {
   363→                this.position.add(this.velocity.clone().multiplyScalar(delta));
   364→                if (this.mesh) this.mesh.position.copy(this.position);
   365→            }
   366→            destroy() {
   367→                this.alive = false;
   368→                if (this.mesh) game.scene.remove(this.mesh);
   369→            }
   370→        }

   372→        class Enemy extends Entity {
   373→            constructor(position, level) {
   374→                super(position, 'enemy');
   375→                this.health = 50 + (level * 10);
   376→                this.maxHealth = this.health;
   377→                this.speed = 3 + (level * 0.5);
   378→                this.level = level;
   379→                this.lastShot = 0;
   380→                this.weapon = this.getRandomWeapon(level);
   381→                this.createMesh();
   382→            }
   383→            getRandomWeapon(level) {
   384→                const weapons = ['pistol', 'smg', 'rifle', 'shotgun'];
   385→                if (level >= 5) weapons.push('sniper');
   386→                if (level >= 10) weapons.push('minigun');
   387→                return weapons[Math.floor(Math.random() * weapons.length)];
   388→            }
   389→            createMesh() {
   390→                this.mesh = new THREE.Group();
   391→                const body = createDoodleMesh(new THREE.BoxGeometry(0.6, 1.4, 0.4), 0xe74c3c);
   392→                body.position.y = 0.7; body.userData.isBody = true;
   393→                const head = createDoodleMesh(new THREE.BoxGeometry(0.4, 0.4, 0.4), 0xffdbac);
   394→                head.position.y = 1.6; head.userData.isHead = true;
   395→                const armGeo = new THREE.BoxGeometry(0.2, 0.8, 0.2);
   396→                const leftArm = createDoodleMesh(armGeo, 0xe74c3c); leftArm.position.set(-0.4, 0.9, 0);
   397→                const rightArm = createDoodleMesh(armGeo, 0xe74c3c); rightArm.position.set(0.4, 0.9, 0);
   398→                const legGeo = new THREE.BoxGeometry(0.25, 0.9, 0.25);
   399→                const leftLeg = createDoodleMesh(legGeo, 0x34495e); leftLeg.position.set(-0.2, -0.45, 0);
   400→                const rightLeg = createDoodleMesh(legGeo, 0x34495e); rightLeg.position.set(0.2, -0.45, 0);
   401→                const gun = createDoodleMesh(new THREE.BoxGeometry(0.1, 0.1, 0.5), 0x2c3e50);
   402→                gun.position.set(0.5, 1.0, 0.3);
   403→                this.mesh.add(body, head, leftArm, rightArm, leftLeg, rightLeg, gun);
   404→                this.mesh.position.copy(this.position);
   405→                this.bbox = new AABB(new THREE.Vector3(-0.3, 0, -0.3), new THREE.Vector3(0.3, 2.0, 0.3));
   406→                game.scene.add(this.mesh);
   407→            }
   408→            takeDamage(damage, isHeadshot) {
   409→                const finalDamage = isHeadshot ? damage * 3 : damage;
   410→                this.health -= finalDamage;
   411→                if (this.mesh) {
   412→                    this.mesh.children.forEach(child => {
   413→                        if (child.userData && child.userData.isDoodle) {
   414→                            child.position.x += (Math.random() - 0.5) * 0.1;
   415→                        }
   416→                    });
   417→                }
   418→                if (this.health <= 0) { this.die(); return true; }
   419→                return false;
   420→            }
   421→            die() {
   422→                this.destroy();
   423→                game.player.killStreak++;
   424→                game.player.kills++;
   425→                game.updateScore(50);
   426→                if (Math.random() < 0.6) game.spawnPickup(this.position.clone(), 'ammo');
   427→                if (Math.random() < 0.2) game.spawnPickup(this.position.clone(), 'health');
   428→                if (Math.random() < 0.15) {
   429→                    const weapons = ['pistol', 'smg', 'rifle', 'shotgun'];
   430→                    if (this.level >= 5) weapons.push('sniper');
   431→                    game.spawnPickup(this.position.clone(), 'weapon', weapons[Math.floor(Math.random() * weapons.length)]);
   432→                }
   433→                if (this.level >= 10 && Math.random() < CONFIG.DRONE_DROP_CHANCE) game.spawnPickup(this.position.clone(), 'drone');
   434→                if (this.level >= 15 && Math.random() < CONFIG.DRAGON_DROP_CHANCE) game.spawnPickup(this.position.clone(), 'dragon');
   435→            }
   436→            update(delta, playerPos) {
   437→                super.update(delta);
   438→                const direction = new THREE.Vector3();
   439→                direction.subVectors(playerPos, this.position);
   440→                direction.y = 0;
   441→                const distance = direction.length();
   442→                direction.normalize();
   443→                if (distance > 5) this.velocity.copy(direction.multiplyScalar(this.speed));
   444→                else this.velocity.set(0, 0, 0);
   445→                if (this.mesh) this.mesh.lookAt(playerPos.x, this.position.y + 1, playerPos.z);
   446→                const now = Date.now();
   447→                if (distance < 25 && now - this.lastShot > WEAPONS[this.weapon].fireRate * 2) {
   448→                    this.shoot(playerPos);
   449→                    this.lastShot = now;
   450→                }
   451→            }
   452→            shoot(targetPos) {
   453→                const weapon = WEAPONS[this.weapon];
   454→                const direction = new THREE.Vector3();
   455→                direction.subVectors(targetPos, this.position);
   456→                direction.y += 0.5;
   457→                direction.normalize();
   458→                direction.x += (Math.random() - 0.5) * weapon.spread;
   459→                direction.z += (Math.random() - 0.5) * weapon.spread;
   460→                const bullet = new Bullet(this.position.clone().add(new THREE.Vector3(0, 1.2, 0)), direction, weapon.damage * 0.7, weapon.bulletSpeed, 'enemy');
   461→                game.bullets.push(bullet);
   462→            }
   463→        }

   465→        class Bullet extends Entity {
   466→            constructor(position, direction, damage, speed, owner) {
   467→                super(position, 'bullet');
   468→                this.direction = direction.clone().normalize();
   469→                this.velocity = this.direction.multiplyScalar(speed);
   470→                this.damage = damage;

   871→            shoot() {
   872→                if (this.isPaused || this.isGameOver || !this.controls.isLocked) return;
   873→                const weaponType = this.player.weapons[this.player.currentWeapon];
   874→                if (!weaponType) return;
   875→                const weapon = WEAPONS[weaponType];
   876→                const now = Date.now();
   877→                if (now - this.lastShot < weapon.fireRate) return;
   878→                if (weapon.ammo <= 0) { this.showNotification('⚠️ Sin munición - Presiona R'); return; }
   879→                this.lastShot = now;
   880→                weapon.ammo--;
   881→                const pellets = weapon.pellets || 1;
   882→                for(let i = 0; i < pellets; i++) {
   883→                    const direction = new THREE.Vector3(0, 0, -1);
   884→                    direction.applyQuaternion(this.camera.quaternion);
   885→                    direction.x += (Math.random() - 0.5) * weapon.spread;
   886→                    direction.y += (Math.random() - 0.5) * weapon.spread;
   887→                    direction.z += (Math.random() - 0.5) * weapon.spread;
   888→                    direction.normalize();
   889→                    this.bullets.push(new Bullet(this.player.position.clone(), direction, weapon.damage, weapon.bulletSpeed, 'player'));
   890→                }
   891→                this.viewModel.position.z += 0.05;
   892→                setTimeout(() => { this.viewModel.position.z = -0.5; }, 50);
   893→                this.updateHUD();
   894→            }

   945→            update(delta) {
   946→                if (this.isPaused || this.isGameOver) return;
   947→                this.updatePlayer(delta);
   948→                for(let i = this.bullets.length - 1; i >= 0; i--) {
   949→                    this.bullets[i].update(delta);
   950→                    if (!this.bullets[i].alive) { this.bullets.splice(i, 1); continue; }
   951→                    this.checkBulletCollisions(this.bullets[i]);
   952→                }
   953→                for(let i = this.enemies.length - 1; i >= 0; i--) {
   954→                    this.enemies[i].update(delta, this.player.position);
   955→                    if (!this.enemies[i].alive) { this.enemies.splice(i, 1); continue; }
   956→                    if (this.enemies[i].position.distanceTo(this.player.position) < 1.5) this.damagePlayer(5);
   957→                }
   958→                for(let i = this.pickups.length - 1; i >= 0; i--) {
   959→                    this.pickups[i].update(delta);
   960→                    if (!this.pickups[i].alive) { this.pickups.splice(i, 1); continue; }
   961→                    if (this.pickups[i].position.distanceTo(this.player.position) < 2) { this.pickups[i].collect(); this.pickups.splice(i, 1); }
   962→                }
   963→                for(let i = this.drones.length - 1; i >= 0; i--) { this.drones[i].update(delta); if (!this.drones[i].alive) this.drones.splice(i, 1); }
   964→                for(let i = this.dragons.length - 1; i >= 0; i--) { this.dragons[i].update(delta); if (!this.dragons[i].alive) this.dragons.splice(i, 1); }
   965→                for(let i = this.greenMobs.length - 1; i >= 0; i--) { this.greenMobs[i].update(delta); if (!this.greenMobs[i].alive) this.greenMobs.splice(i, 1); }
   966→                if (Math.random() < CONFIG.GREEN_MOB_CHANCE) this.greenMobs.push(new GreenMob(new THREE.Vector3((Math.random() - 0.5) * 40, 0, (Math.random() - 0.5) * 40)));
   967→                if (this.enemies.length === 0 && this.isShooting === false) { // Simple check to ensure wave is truly done
   968→                    // Defer level up slightly to let last bullets resolve
   969→                    setTimeout(() => {
   970→                        if(this.enemies.length === 0 && !this.isGameOver) {
   971→                            this.currentLevel++;
   972→                            if (this.currentLevel > 20) this.victory();
   973→                            else this.startWave();
   974→                        }
   975→                    }, 1000);
   976→                }
   977→                this.scene.traverse((obj) => {
   978→                    if (obj.userData && obj.userData.isDoodle && obj.userData.wireframe) {
   979→                        const t = Date.now() * obj.userData.wobbleSpeed + obj.userData.wobbleOffset;
   980→                        obj.userData.wireframe.rotation.z = Math.sin(t) * 0.02;
   981→                    }
   982→                });
   983→                this.updateHUD();
   984→            }
   985→            
   986→            updatePlayer(delta) {
   987→                const speed = CONFIG.PLAYER_SPEED;
   988→                const velocity = this.player.velocity;
   989→                velocity.y -= CONFIG.GRAVITY * delta;
   990→                const direction = new THREE.Vector3();
   991→                if (this.moveState.forward) direction.z -= 1;
   992→                if (this.moveState.backward) direction.z += 1;
   993→                if (this.moveState.left) direction.x -= 1;
   994→                if (this.moveState.right) direction.x += 1;
   995→                direction.normalize();
   996→                if (direction.length() > 0) {
   997→                    const moveX = direction.x * speed * delta;
   998→                    const moveZ = direction.z * speed * delta;
   999→                    const rotatedMove = new THREE.Vector3(moveX, 0, moveZ);
  1000→                    rotatedMove.applyAxisAngle(new THREE.Vector3(0, 1, 0), this.camera.rotation.y);
  1001→                    const newPos = this.player.position.clone();
  1002→                    newPos.x += rotatedMove.x; newPos.z += rotatedMove.z;
  1003→                    let canMove = true;
  1004→                    const playerBBox = new AABB(new THREE.Vector3(newPos.x - 0.3, newPos.y - 1.6, newPos.z - 0.3), new THREE.Vector3(newPos.x + 0.3, newPos.y + 0.4, newPos.z + 0.3));
  1005→                    for(const obstacle of this.obstacles) {
  1006→                        if (playerBBox.intersects(obstacle.bbox)) { canMove = false; break; }
  1007→                    }
  1008→                    if (canMove) { this.player.position.x = newPos.x; this.player.position.z = newPos.z; }
  1009→                }
  1010→                this.player.position.y += velocity.y * delta;
  1011→                if (this.player.position.y < 1.6) { this.player.position.y = 1.6; velocity.y = 0; this.player.canJump = true; }
  1012→                this.camera.position.copy(this.player.position);
  1013→                if (this.isShooting) {
  1014→                    const weaponType = this.player.weapons[this.player.currentWeapon];
  1015→                    if (weaponType && WEAPONS[weaponType].automatic) this.shoot();
  1016→                }
  1017→            }
  1018→            
  1019→            checkBulletCollisions(bullet) {
  1020→                for(let i = 0; i < this.enemies.length; i++) {
  1021→                    const enemy = this.enemies[i];
  1022→                    if (!enemy.alive) continue;
  1023→                    const distance = bullet.position.distanceTo(enemy.position);
  1024→                    if (distance < 1.5) {
  1025→                        const headPos = enemy.position.clone(); headPos.y += 1.6;
  1026→                        const isHeadshot = bullet.position.distanceTo(headPos) < 0.5;
  1027→                        if (enemy.takeDamage(bullet.damage, isHeadshot)) this.enemies.splice(i, 1);
  1028→                        bullet.destroy(); return;
  1029→                    }
  1030→                }
  1031→                for(const obstacle of this.obstacles) {
  1032→                    if (obstacle.bbox.contains(bullet.position)) { bullet.destroy(); return; }
  1033→                }
  1034→            }

  1036→            damagePlayer(damage) {
  1037→                let actualDamage = damage;
  1038→                if (this.player.shield > 0) {
  1039→                    const shieldDamage = Math.min(this.player.shield, damage);
  1040→                    this.player.shield -= shieldDamage; actualDamage -= shieldDamage;
  1041→                }
  1042→                this.player.health -= actualDamage;
  1043→                if (this.player.health <= 0) { this.player.health = 0; this.gameOver(); }
  1044→                this.player.killStreak = 0;
  1045→                this.updateHUD();
  1046→            }
