   193→        function createDoodleMesh(geometry, color, scale = 1) {
   194→            const group = new THREE.Group();
   195→            
   196→            // Relleno pastel sólido (Toon para evitar degradados realistas)
   197→            const material = new THREE.MeshToonMaterial({ color: color });
   198→            const mesh = new THREE.Mesh(geometry, material);
   199→            
   200→            // Contorno de tinta negra irregular (EdgesGeometry)
   201→            const edges = new THREE.EdgesGeometry(geometry, 20);
   202→            const lineMaterial = new THREE.LineBasicMaterial({ color: 0x000000, linewidth: 2 });
   203→            const wireframe = new THREE.LineSegments(edges, lineMaterial);
   204→            
   205→            // Escalar ligeramente el wireframe para evitar z-fighting y dar efecto de "doble trazo"
   206→            wireframe.scale.set(1.015, 1.015, 1.015);
   207→            
   208→            group.add(mesh);
   209→            group.add(wireframe);
   210→            group.scale.set(scale, scale, scale);
   211→            
   212→            // Datos para animación de "garabato" (vibración de tinta)
   213→            group.userData = { 
   214→                wireframe: wireframe,
   215→                wobbleSpeed: Math.random() * 0.003 + 0.002,
   216→                wobbleOffset: Math.random() * 100,
   217→                isDoodle: true
   218→            };
   219→            
   220→            return group;
   221→        }

   305→

   442→        function shoot() {
   443→            if (!controls.isLocked || isGameOver) return;
   444→            const now = Date.now();
   445→            if (now - lastShot < 150) return; // Cadencia de tiro
   446→            lastShot = now;

   448→            const bullet = new THREE.Mesh(bulletGeo, bulletMat);
   449→            bullet.position.copy(camera.position);
   450→            
   451→            const vector = new THREE.Vector3(0, 0, -1);
   452→            vector.applyQuaternion(camera.quaternion);
   453→            bullet.userData = { velocity: vector.multiplyScalar(2.0), life: 60 };
   454→            
   455→            scene.add(bullet);
   456→            bullets.push(bullet);
   457→
   458→            // Animación de retroceso del View Model

   496→        let prevTime = performance.now();

   498→        function animate() {
   499→            requestAnimationFrame(animate);
   500→            if (isGameOver) return;
   501→
   502→            const time = performance.now();
   503→            const delta = (time - prevTime) / 1000;
   504→            prevTime = time;
   505→
   506→            if (controls.isLocked) {
   507→                // Física de movimiento
   508→                velocity.x -= velocity.x * 10.0 * delta;
   509→                velocity.z -= velocity.z * 10.0 * delta;
   510→                velocity.y -= 25.0 * delta; // Gravedad

   514→                direction.normalize();

   534→                }

   539→                    viewModel.rotation.x = -recoil * 1.5;

   543→                    viewModel.rotation.x = 0;

   550→                    b.userData.life--;
   551→

   553→                        scene.remove(b);
   554→                        bullets.splice(i, 1);
   555→                        continue;
   556→                    }
   557→
   558→                    // Colisión con jugadores remotos (simulación de hit)
   559→                    for (let j = remotePlayers.length - 1; j >= 0; j--) {
   560→                        const rp = remotePlayers[j];
   561→                        if (b.position.distanceTo(rp.group.position) < 1.0) {
   562→                            rp.health -= 34; // 3 disparos para matar

   564→                            bullets.splice(i, 1);
   565→

   568→                                remotePlayers.splice(j, 1);

   573→                                setTimeout(() => {
   574→                                    if (!isGameOver) remotePlayers.push(new RemotePlayer(Date.now()));
   575→                                }, 2000);
   576→                            }
   577→                            break;
   578→                        }
   579→                    }
   580→                }
   581→
   582→                // Actualizar jugadores remotos (Simulación de red)
   583→                remotePlayers.forEach(rp => rp.update());
   584→                document.getElementById('playerCount').innerText = remotePlayers.length + 1;
   585→
   586→                // Modo Supervivencia: Oleadas (si aplica)
   587→                if (gameMode === 'solo' && remotePlayers.length < 3 + Math.floor(score/50)) {
   588→                    if (Math.random() < 0.02) remotePlayers.push(new RemotePlayer(Date.now()));
   589→                }
   590→            }
   591→
   592→            // Animación global de "garabato" en todos los objetos doodle
   593→            scene.traverse((obj) => {
   594→                if (obj.userData && obj.userData.isDoodle) {

   597→                    obj.userData.wireframe.scale.setScalar(1.015 + Math.sin(t*2) * 0.005);
   598→                }
   599→            });
   600→
   601→            renderer.render(scene, camera);
